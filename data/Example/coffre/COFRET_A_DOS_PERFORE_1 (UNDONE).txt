# Example I need to make a powder-coated steel enclosure. The enclosure should have dimensions of 200 mm in length, 100 mm in depth, and 250 mm in height. It must include folds for closing the top and bottom of the enclosure. Also, add 20 mm folds along the 250 mm sides.
# On the base, there should be four Ø10 holes in the corners, each 30 mm from the edges.
# Additionally, add two Ø40 holes centered in width — the first one positioned 50 mm from the bottom edge, and the second one 85 mm above it (center-to-center distance).
# The enclosure has a thickness of 1.5 mm.
# Shape type: capot
# -*- coding: utf-8 -*-
"""
Box Shape with Additional Bends - FreeCAD Script
Creates a tub shape with additional bends, fusing, and filleting operations.
"""

import sys
import os
import FreeCAD as App
import Import
import Mesh
import MeshPart
try:
    from sheetmetal import SheetMetalCmd
except ImportError:
    App.Console.PrintError("SheetMetal workbench is not installed. Please install it from the Addon Manager.\n")


# FreeCadUtil import (mandatory for all scripts)
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

doc = App.newDocument("Box_Shape_With_Additional_Bend")

thickness = 2.0
bend_radius = thickness # Default inner bend radius
length = 200.0  # X-dimension
width = 250.0   # Y-dimension
height = 100.0

# Additional bend parameters
additional_bend_length = 20.0
additional_bend_angle = 90.0


tub_obj = Part.makeTub(
    thickness=thickness,
    bend_radius=bend_radius,
    dim_y=width,
    dim_x=length,
    height=height,
    type="closed"
)

doc.recompute()

# Add bends using the AddBend function
bent_obj = AddInwardBend(tub_obj, additional_bend_length, additional_bend_angle, bend_radius, axis='Y')

doc.recompute()

# Create holes as specified in the requirements
# Four Ø10 holes in corners, 30mm from edges
hole_diameter_small = 10.0
hole_diameter_large = 40.0
corner_offset = 30.0

# Calculate positions for corner holes (centered at 0,0)
corner_holes = []
corner_positions = [
    (-length/2 + corner_offset, -width/2 + corner_offset),  # Bottom-left
    (length/2 - corner_offset, -width/2 + corner_offset),   # Bottom-right
    (length/2 - corner_offset, width/2 - corner_offset),    # Top-right
    (-length/2 + corner_offset, width/2 - corner_offset)    # Top-left
]

for x, y in corner_positions:
    hole = Part.makeCylinder(
        hole_diameter_small/2,
        thickness + 10,
        App.Vector(x, y, -5),
        App.Vector(0, 0, 1)
    )
    corner_holes.append(hole)

# Two Ø80 holes centered in width
large_holes = []
# First hole: 50mm from bottom edge
hole1_y = -width/2 + 50.0
# Second hole: 85mm above the first (center-to-center)
hole2_y = hole1_y + 85.0

large_hole_positions = [
    (0, hole1_y),  # Centered in width, 50mm from bottom
    (0, hole2_y)   # Centered in width, 85mm above first hole
]

for x, y in large_hole_positions:
    hole = Part.makeCylinder(
        hole_diameter_large/2,
        thickness + 10,
        App.Vector(x, y, -5),
        App.Vector(0, 0, 1)
    )
    large_holes.append(hole)

# Combine all holes
all_holes = corner_holes + large_holes

# Create a compound of all holes
if all_holes:
    holes_compound = Part.makeCompound(all_holes)

    # Cut holes from the bent object using Shape
    cut_shape = bent_obj.Shape.cut(holes_compound)

    # Create a new Part object with the cut shape
    final_obj = doc.addObject("Part::Feature", "Perforated_Box")
    final_obj.Shape = cut_shape
else:
    final_obj = bent_obj

doc.recompute()

# Export directory handling
sanitized_title = "Box_shape_with_additional_bend"
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([final_obj], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(cut_shape, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    cut_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)


# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [final_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)

# Final recompute (optional)
doc.recompute()
