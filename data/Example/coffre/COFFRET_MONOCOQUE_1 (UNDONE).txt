# Example I would like a folded enclosure with a base measuring 304x204 mm and a thickness of 2 mm, featuring a central rectangular cutout of 100x50 mm, with the 100 mm side oriented along the 304 mm edge. There are four vertical folds of 102 mm on each side, and an additional four folds of 25 mm at the ends of these folds, with 45° corner cuts to avoid collisions.
# Shape type: capot
# # -*- coding: utf-8 -*-
"""
Box Shape with Additional Bends - FreeCAD Script
Creates a tub shape with additional bends, fusing, and filleting operations.
"""

import sys
import os
import FreeCAD as App
import Import
import Mesh
import MeshPart
try:
    from sheetmetal import SheetMetalCmd
except ImportError:
    App.Console.PrintError("SheetMetal workbench is not installed. Please install it from the Addon Manager.\n")


# FreeCadUtil import (mandatory for all scripts)
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

doc = App.newDocument("Box_Shape_With_Additional_Bend")

thickness = 2.0
bend_radius = thickness # Default inner bend radius
length = 204.0  # X-dimension
width = 304.0   # Y-dimension
height = 102

# Additional bend parameters
additional_bend_length = 25.0
additional_bend_angle = 90.0


tub_obj = Part.makeTub(
    thickness=thickness,
    bend_radius=bend_radius,
    dim_y=width,
    dim_x=length,
    height=height,
    type="closed"
)

doc.recompute()

# Uss this function to Creates additional inward bend on four high side walls of makeTub shape with automatic 45° mitered cuts at corners
bent_obj = AddInwardBendExtended(tub_obj, additional_bend_length, additional_bend_angle, bend_radius)

doc.recompute()

# Create central rectangular cutout (100x50 mm, with 100mm side along the 304mm edge)
cutout_length = 100.0  # Along Y-dimension (304mm edge)
cutout_width = 50.0    # Along X-dimension (204mm edge)

# Position cutout at center (0,0) - center the cutout around origin
cutout_x = -cutout_width / 2   # Center along X
cutout_y = -cutout_length / 2  # Center along Y

# Create cutout box - make it slightly thicker than the base to ensure clean cut
cutout_box = Part.makeBox(cutout_width, cutout_length, thickness + 2.0)
cutout_box.Placement = App.Placement(App.Vector(cutout_x, cutout_y, -0.5), App.Rotation())

# Add cutout box to document for visualization (optional)
cutout_obj = doc.addObject("Part::Feature", "Cutout")
cutout_obj.Shape = cutout_box

# Perform the cut operation using shape.cut() method
cut_shape = bent_obj.Shape.cut(cutout_box)
final_obj = doc.addObject("Part::Feature", "Box_with_Cutout")
final_obj.Shape = cut_shape

doc.recompute()

# Export directory handling
sanitized_title = "Box_shape_with_additional_bend"
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([final_obj], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(cutout_box, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    cutout_box,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)


# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [final_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)

# Final recompute (optional)
doc.recompute()
