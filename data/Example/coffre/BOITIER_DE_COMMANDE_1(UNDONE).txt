# Example I would like a wall-mounted control box made of 1.5 mm thick steel sheet, with dimensions of 300 mm in width, 253 mm in height, and 150 mm in depth.
# I want a front door with hinges on the left and a lockable latch for closure.
# It should have 4 holes with a diameter of 22 mm for push buttons, arranged in a horizontal line centered on the front face, with a spacing of 40 mm between centers.
# Shape type: capot

# -*- coding: utf-8 -*-
"""
Box Shape with Additional Bends - FreeCAD Script
Creates a tub shape with additional bends, fusing, and filleting operations.
"""

import sys
import os
import FreeCAD as App
import Import
import Mesh
import MeshPart
try:
    from sheetmetal import SheetMetalCmd
except ImportError:
    App.Console.PrintError("SheetMetal workbench is not installed. Please install it from the Addon Manager.\n")


# FreeCadUtil import (mandatory for all scripts)
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

doc = App.newDocument("Box_Shape_With_Additional_Bend")

thickness = 2.0
bend_radius = thickness # Default inner bend radius
length = 300  # X-dimension
width = 253  # Y-dimension
height = 150

# Additional bend parameters
additional_bend_length = 20.0
additional_bend_angle = 90.0


tub_obj = Part.makeTub(
    thickness=thickness,
    bend_radius=bend_radius,
    dim_y=width,
    dim_x=length,
    height=height,
    type="closed"
)

doc.recompute()

# Add bends using the AddBend function
bent_obj = AddInwardBend(tub_obj, additional_bend_length, additional_bend_angle, bend_radius, axis='X')

doc.recompute()

# Create holes as requested
# 4 push button holes (Ø22mm) on base (bottom face), horizontal line, 40mm spacing
button_hole_diameter = 22
button_spacing = 40

# Calculate positions for 4 buttons centered horizontally on base
# Total span = 3 * spacing = 120mm, so buttons at -60, -20, +20, +60 from center
button_positions_x = [-60, -20, 20, 60]
button_y = 0  # Center on Y-axis
button_z = 0  # On the base (bottom face)

# Create push button holes (vertical cylinders through the base)
button_holes = []
for x_pos in button_positions_x:
    hole = Part.makeCylinder(
        button_hole_diameter/2,
        thickness + 10,
        App.Vector(x_pos, button_y, button_z - 5),
        App.Vector(0, 0, 1)
    )
    button_holes.append(hole)

# Combine all holes (only button holes now)
all_holes = button_holes
combined_holes = all_holes[0]
for hole in all_holes[1:]:
    combined_holes = combined_holes.fuse(hole)

# Cut holes from the bent object
final_shape = bent_obj.Shape.cut(combined_holes)
final_obj = doc.addObject("Part::Feature", "ControlBoxWithHoles")
final_obj.Shape = final_shape

doc.recompute()

# Export directory handling
sanitized_title = "Box_shape_with_additional_bend"
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([final_obj], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(final_shape, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    final_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)


# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [final_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)

# Final recompute (optional)
doc.recompute()
