#Example Make a circular plate with a diameter of 200 mm and a thickness of 3 mm.
# Shape type: Sheet-Circular

import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# Define the sanitized_title variable
sanitized_title = "Circular_Plate_D200_T3"

# Document Setup
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# Parameters
diameter = 200.0
thickness = 3.0
radius = diameter / 2.0

# Create the circular plate using Part.makeCylinder
circular_shape = Part.makeCylinder(radius, thickness, App.Vector(0, 0, 0), App.Vector(0, 0, 1))

# Register final object
plate_obj = doc.addObject("Part::Feature", "FinishedPlate")
plate_obj.Label = sanitized_title
plate_obj.Shape = circular_shape
doc.recompute()

# Export directory handling
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([plate_obj], step_path)

# Geometry analysis and export
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json
geometry_result = analyze_geometry(circular_shape, material="{mapped_material}")

geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    circular_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)

# Export OBJ with separate meshes for each face
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")
mesh_objects = []
face_counter = 1

for part in [plate_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape            = face,
            LinearDeflection = 0.1,
            AngularDeflection= 0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh  = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)
doc.recompute()

#Example Make a quarter ring / quarter annulus (anneau, couronne circulaire, vành khuyên) — a hollow curved band, NOT a solid disc — with outer diameter of 400 mm, band width (largeur de la bande, radial width between outer and inner edge) of 50 mm, cut to a 90-degree quarter arc, thickness of 4 mm. This has a hole cut out of the middle (hollow band); it is different from a solid quarter-disc/quarter-circle plate which has no inner hole.
# Shape type: Sheet-Circular

import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# Define the sanitized_title variable
sanitized_title = "Quarter_Ring_D400_BW50_T4"

# Document Setup
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# Parameters
outer_diameter = 400.0
thickness = 4.0
band_width = 50.0   # radial width of the ring band (this is what the customer calls "largeur")
arc_angle = 90.0     # 90 = quarter ring, 180 = half ring, 360 = full ring (donut)
outer_radius = outer_diameter / 2.0

# GEOMETRY GUARD: band_width must be strictly smaller than outer_radius,
# otherwise inner_radius would be <= 0 and there would be no ring left to cut
assert band_width < outer_radius, "band_width must be smaller than outer_radius"
inner_radius = outer_radius - band_width

# Build the ring sector: outer sector minus inner sector of the same arc_angle
outer_sector = Part.makeCylinder(outer_radius, thickness, App.Vector(0, 0, 0), App.Vector(0, 0, 1), arc_angle)
inner_sector = Part.makeCylinder(inner_radius, thickness, App.Vector(0, 0, 0), App.Vector(0, 0, 1), arc_angle)
circular_shape = outer_sector.cut(inner_sector)

# Register final object
plate_obj = doc.addObject("Part::Feature", "FinishedPlate")
plate_obj.Label = sanitized_title
plate_obj.Shape = circular_shape
doc.recompute()

# Export directory handling
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([plate_obj], step_path)

# Geometry analysis and export
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json
geometry_result = analyze_geometry(circular_shape, material="{mapped_material}")

geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    circular_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)

# Export OBJ with separate meshes for each face
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")
mesh_objects = []
face_counter = 1

for part in [plate_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape            = face,
            LinearDeflection = 0.1,
            AngularDeflection= 0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh  = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)
doc.recompute()
