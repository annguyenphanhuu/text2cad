# Example I want an oblong access hatch measuring Ø400 x 200, 3mm thick with two Ø20mm holes with a 200mm center distance centered on the major axis of the oblong.
# Shape type: plate
import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# Define the sanitized_title variable  
sanitized_title = "TrappeAcces_Oblong_400x200x3_2TrousD20"   # DO NOT CHANGE THIS VALUE

# Document Setup
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# Parameters (mm)
plate_length_total = 400.0   # Overall X dimension after fusion
plate_width        = 200.0   # Y dimension
plate_thickness    =   3.0   # Z dimension

# Create oblong plate using makeOblong function

OblongPlate = Part.makeOblong(
    length=plate_length_total,    # Total length including circular ends (400mm)
    width=plate_width,            # Width (diameter of end circles) (200mm)
    height=plate_thickness,       # Thickness (3mm)
    pnt=App.Vector(0, 0, 0),      # Start from origin
    dir=App.Vector(0, 0, 1)       # Extrude along positive Z-axis
)

# Create two D20 holes (individually, <4 so NO loops)
hole_diam   = 20.0
hole_radius = hole_diam / 2.0
hole_depth  = plate_thickness + 2.0

# Hole center coordinates (entrax 200 mm, centered)
center_y = plate_width / 2.0
hole1_center_x = (plate_length_total / 2.0) - 100.0   # 400/2 - 100 = 100
hole2_center_x = (plate_length_total / 2.0) + 100.0   # 400/2 + 100 = 300

# Hole 1
Hole1 = Part.makeCylinder(
    hole_radius,
    hole_depth,
    App.Vector(hole1_center_x, center_y, -1.0),
    App.Vector(0, 0, 1)
)

# Hole 2
Hole2 = Part.makeCylinder(
    hole_radius,
    hole_depth,
    App.Vector(hole2_center_x, center_y, -1.0),
    App.Vector(0, 0, 1)
)

# Cut holes from plate
PlateHole1 = OblongPlate.cut(Hole1)
TrappeFinale = PlateHole1.cut(Hole2)

# Register final object
plate_obj = doc.addObject("Part::Feature", "TrappeAcces")
plate_obj.Label = sanitized_title
plate_obj.Shape = TrappeFinale
doc.recompute()

# Export directory handling
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([plate_obj], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(TrappeFinale, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    TrappeFinale,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)


# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [plate_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)

# Final recompute (optional)
doc.recompute()
