#Example I want a guiding base with positioning pins featuring multiple notches. A 400x400x5 plate is required, with 12 Ø6.2 holes arranged in 2 rows of 6, spaced 150mm apart and centered. 
#The notches should be spaced every 40mm, giving a pitch of 200mm between the two ends. Add Ø6 pins of length 20mm in each hole, protruding 20mm on one side. 
#Add 4 M8 holes in each corner at 40mm from the edges.
# Shape type: plate
import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart
import Draft

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# Define the sanitized_title variable  
sanitized_title = "Socle_de_guidage_4004005_mm_avec_perages_62_pions_620_et_4_perages_M8"   # DO NOT CHANGE THIS VALUE

# Document Setup
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# Parameters (mm)
plate_length = 400.0  # X dimension (length)
plate_width  = 400.0  # Y dimension (width)
plate_thick  =   5.0  # Z dimension (thickness)

# Hole specifications
hole_d62_radius = 6.2 / 2.0  # D6.2 hole radius
hole_m8_radius  = 8.0 / 2.0  # M8 hole radius

# Pin specifications
pin_radius      = 6.0 / 2.0   # D6 pin radius
pin_length      = 20.0        # Pin protrusion height above plate surface

# X positions: 2 columns centered with 150mm spacing
# Centering logic: Calculate equal margins on both sides to center columns
column_spacing_x = 150.0
num_columns = 2

total_span_x = (num_columns - 1) * column_spacing_x
remaining_x = plate_length - total_span_x
margin_x = remaining_x / 2.0
first_x = margin_x

x_positions = [first_x + i * column_spacing_x for i in range(num_columns)]

# Y positions: 6 holes per column, centered with 40mm spacing
# Centering logic: Calculate equal margins to center holes vertically
y_spacing = 40.0
num_holes = 6

total_span_y = (num_holes - 1) * y_spacing
remaining_y = plate_width - total_span_y
margin_y = remaining_y / 2.0
first_y = margin_y

y_positions = [first_y + i * y_spacing for i in range(num_holes)]

# Corner M8 holes at 40mm from edges
corner_offset = 40.0

corner_positions = [
    (corner_offset,                 corner_offset),                 # Bottom-left
    (plate_length - corner_offset, corner_offset),                 # Bottom-right
    (corner_offset,                 plate_width - corner_offset),  # Top-left
    (plate_length - corner_offset, plate_width - corner_offset),  # Top-right
]

# Create the base plate
plate = Part.makeBox(plate_length, plate_width, plate_thick, App.Vector(0, 0, 0))

# Drill D6.2 holes (12 holes total) - Batch operation
# Pattern: 2 columns × 6 holes = 12 holes
# Use batch compound operation for efficiency when hole count >= 4
holes_62 = []
for x in x_positions:
    for y in y_positions:
        cyl = Part.makeCylinder(
            hole_d62_radius,
            plate_thick + 2.0,  # Extra depth ensures complete cut
            App.Vector(x, y, plate_thick),
            App.Vector(0, 0, -1)
        )
        holes_62.append(cyl)

compound_62 = Part.makeCompound(holes_62)
plate = plate.cut(compound_62)

# Drill M8 corner holes (4 holes) - Batch operation
corner_holes = []
for (x, y) in corner_positions:
    cyl = Part.makeCylinder(
        hole_m8_radius,
        plate_thick + 2.0,  # Extra depth ensures complete cut
        App.Vector(x, y, plate_thick),
        App.Vector(0, 0, -1)
    )
    corner_holes.append(cyl)

compound_m8 = Part.makeCompound(corner_holes)
plate = plate.cut(compound_m8)

# Add D6x20 pins protruding 20mm on top
# Pattern: 2 columns × 6 pins = 12 pins
# Pins positioned at same locations as D6.2 holes
pins = []
for x in x_positions:
    for y in y_positions:
        pin = Part.makeCylinder(
            pin_radius,
            pin_length,
            App.Vector(x, y, plate_thick),
            App.Vector(0, 0, 1)
        )
        pins.append(pin)

compound_pins = Part.makeCompound(pins)
final_shape = plate.fuse(compound_pins)

# Register final object
part_obj = doc.addObject("Part::Feature", "Guiding_Base")
part_obj.Label = sanitized_title
part_obj.Shape = final_shape
doc.recompute()

# Export directory handling
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# Export STEP
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([part_obj], step_path)


# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(final_shape, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    final_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)

# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [part_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)

# Final recompute (optional)
doc.recompute()
