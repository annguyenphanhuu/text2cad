# Example a support for a small air extractor, made of galvanized steel sheet DX51D+Z with a thickness of 2 mm.
# It consists of a vertical plate 300 mm wide and 200 mm high, with a 50 mm return flange at the bottom, bent at 90° to form a mounting bracket. The internal bend radius is 2 mm.
# On the vertical face, I require a central circular cut-out Ø150 mm for duct passage, with four Ø6 mm mounting holes around the cut-out, spaced at 90°, and located 10 mm from the edge of the cut-out.
# On the base flange, provide two Ø8 mm holes positioned 30 mm from the edges.
# Shape type: L-bracket

# -*- coding: utf-8 -*-
import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# Define the sanitized_title variable
sanitized_title = "L_shaped_sheet_metal_support_for_small_air_extractor"   # DO NOT CHANGE THIS VALUE

# Document Setup
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# Parameters (mm)
# Geometry
# 1. Create the L-shaped bracket (Vertical plate + horizontal base flange)
dim_x          =  50.0        # Base length (X dimension) - horizontal leg (leg1)
dim_y          = 300.0        # Depth/Width of plate (Y dimension)
thickness      =   2.0        # Sheet thickness
flange_height  = 200.0        # Vertical leg height/hauteur (Z dimension)
bend_angle_deg =  90.0        # Bend angle
bend_radius    = thickness    # Default inner bend radius

# Horizontal part: 50x300 (base flange)
# Vertical part: 200x300 (vertical plate)
bracket_shape = Part.makeLShape(
    dim_x=dim_x,
    dim_y=dim_y,
    thickness=thickness,
    flange_height=flange_height,
    bend_angle_deg=bend_angle_deg,
    bend_radius=bend_radius
)

# 2. Add ONE central circular cut-out Ø150 mm on the vertical plate
# Result: One duct hole centered on the vertical face

# Duct hole parameters
duct_diameter   = 150.0
duct_radius     = duct_diameter / 2.0          # R = 75mm
duct_center_y   = dim_y / 2.0                  # Centered in width: Y = 300/2 = 150mm (CONSTANT)
duct_center_z   = flange_height / 2.0          # Centered in height: Z = 200/2 = 100mm (CONSTANT)

# Central duct cut-out (Ø150mm)
bracket_shape = add_hole_leg2(
    bracket_shape,
    duct_radius,
    duct_center_y,                            
    duct_center_z,                            
    bend_angle_deg,
    dim_y,
    thickness,
    flange_height,
    bend_radius
)

# 3. Add FOUR mounting holes Ø6 mm around the duct cut-out on the vertical plate (leg2)
# Holes at 0°, 90°, 180°, 270° — located 10 mm from the edge of the cut-out (Ø150mm)
# Result: Radial distance from duct center = duct_radius + 10mm = 85mm

# Fixing hole parameters
fix_hole_diameter    =   6.0
fix_hole_radius      = fix_hole_diameter / 2.0  # R = 3mm
fix_hole_offset_edge =  10.0                     # Distance from duct edge (mm)
radial_dist          = duct_radius + fix_hole_offset_edge  # = 75 + 10 = 85mm from duct center

# 3.1 Hole at 0° (right of duct center)  — Y = 150+85=235mm, Z = 100mm
# 3.2 Hole at 90° (above duct center)   — Y = 150mm,     Z = 100+85=185mm
# 3.3 Hole at 180° (left of duct center) — Y = 150-85=65mm,  Z = 100mm
# 3.4 Hole at 270° (below duct center)  — Y = 150mm,     Z = 100-85=15mm
for ang in (0, 90, 180, 270):
    rad = math.radians(ang)
    # Calculate offset in Y-Z plane (on the vertical plate)
    offset_y = radial_dist * math.cos(rad)
    offset_z = radial_dist * math.sin(rad)

    hole_y = duct_center_y + offset_y
    hole_z = duct_center_z + offset_z

    bracket_shape = add_hole_leg2(
        bracket_shape,
        fix_hole_radius,
        hole_y,
        hole_z,
        bend_angle_deg,
        dim_y,
        thickness,
        flange_height,
        bend_radius
    )

# 4. Add TWO mounting holes Ø8 mm on the base flange (horizontal leg - leg1)
# Holes positioned 30 mm from the edges (Y-direction), centered in base length (X-direction)

# Base hole parameters
base_hole_diameter   =   8.0
base_hole_radius     = base_hole_diameter / 2.0    # R = 4mm
base_hole_edge_y     =  30.0                       # Distance from side edge (Y-direction)
base_hole_x_center   = dim_x / 2.0                # Centered in base length: X = 50/2 = 25mm (CONSTANT)

# 4.1 Hole 1: 30 mm from bottom edge
# Result: X = 25mm (centered), Y = 30mm
bracket_shape = add_hole_leg1(
    bracket_shape,
    base_hole_radius,
    base_hole_x_center,                            # X = 25mm (centered)
    base_hole_edge_y,                              # Y = 30mm from bottom edge
    dim_y,
    thickness,
    dim_x
)

# 4.2 Hole 2: 30 mm from top edge
# Result: X = 25mm (centered), Y = 300 - 30 = 270mm
bracket_shape = add_hole_leg1(
    bracket_shape,
    base_hole_radius,
    base_hole_x_center,                            # X = 25mm (centered)
    dim_y - base_hole_edge_y,                      # Y = 270mm from bottom edge (30mm from top)
    dim_y,
    thickness,
    dim_x
)

# 5. Register final object
final_bracket_obj = doc.addObject("Part::Feature", "Bracket_Final")
final_bracket_obj.Label = sanitized_title
final_bracket_obj.Shape = bracket_shape
doc.recompute()

# Export directory handling
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([final_bracket_obj], step_path)

# GEOMETRY ANALYSIS
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(bracket_shape, material="{mapped_material}")

# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    bracket_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)

# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [final_bracket_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)

# Final recompute (optional)
doc.recompute()
