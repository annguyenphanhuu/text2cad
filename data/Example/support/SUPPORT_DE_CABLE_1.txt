# Example A flat cable support, cut from a 3 mm thick sheet of stainless steel 304.
# The part should measure 300 mm in length and 40 mm in width.
# I would like to integrate four vertical oblong holes of 10 mm by 20 mm, arranged along the length, spaced evenly every 70 mm, and centered in width.
# In addition, the number 30 should be engraved at the bottom right, with a 0.1 mm deep imprint, measuring 7 x 7 mm.
# Shape type: plate
# -*- coding: utf-8 -*-
import os
import sys
import math
import FreeCAD as App
import Part
import Draft
import Import
import Mesh
import MeshPart

# Mandatory FreeCadUtil import block (DO NOT MODIFY)
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *
# METADATA AND DOCUMENT SETUP
sanitized_title = "Support_Cable_300x40x3_Inox304_4Oblong_Gravure30"   # DO NOT CHANGE
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# PARAMETERS (mm)
# Base plate dimensions
plate_length    = 300.0   # X dimension
plate_width     =  40.0   # Y dimension
plate_thickness =   3.0   # Z dimension
origin          = App.Vector(0.0, 0.0, 0.0)

# CREATE BASE PLATE
base_plate_shape = Part.makeBox(plate_length, plate_width, plate_thickness, origin)

# CREATE OBLONG HOLES  (VERTICAL ORIENTATION  -  length along Y axis)
# Oblong hole parameters (D10 x 20  -> overall length = 20 + 10 = 30)
oblong_width        = 10.0                 # D width  ( now X-width after rotation)
oblong_straight_len = 20.0                 # Straight section length (along Y after rotation)
oblong_total_len    = oblong_straight_len + oblong_width
oblong_depth        = plate_thickness + 2  # Extra depth for robust cut
hole_count          = 4
hole_spacing        = 70.0                 # Center-to-center spacing along X
center_y            = plate_width / 2.0    # Centered in width

# Compute X-centers with equal margins
total_span_x = (hole_count - 1) * hole_spacing
margin_x = (plate_length - total_span_x) / 2.0
x_centers = [margin_x + i * hole_spacing for i in range(hole_count)]

oblong_tools = []
for cx in x_centers:
    # Lower-left corner of oblong bounding box for vertical orientation
    corner_x = cx - oblong_width / 2.0
    corner_y = center_y - oblong_total_len / 2.0
    corner_vec = App.Vector(corner_x, corner_y, -1.0)  # start slightly below
    # Direction vector (0,1,0) => main axis along Y  (vertical slot)
    oblong = Part.makeOblong(
        length = oblong_total_len,
        width  = oblong_width,
        height = oblong_depth,
        pnt    = corner_vec,
        dir    = App.Vector(0, 0, 1) # Z-axis for height
    )
    oblong_tools.append(oblong)

# Cut all oblong holes in one operation (compound)
holes_compound = Part.makeCompound(oblong_tools)
plate_with_holes = base_plate_shape.cut(holes_compound)

# ENGRAVING
# Engraving parameters
engraving_string  = "30"
engraving_depth   = 0.1   # Positive number, extruded downward
char_height       = 7.0

# Create text shape
font_path = "/app/font/boston_traffic.ttf"  # default font path
txt_obj = Draft.makeShapeString(
    String=engraving_string,
    FontFile=font_path,
    Size=char_height,
    Tracking=0
)
doc.recompute()

# Position text at bottom-right of the plate (5 mm margins)
bbox = txt_obj.Shape.BoundBox
txt_width  = bbox.XLength
txt_height = bbox.YLength

offset_x = plate_length - txt_width - 5.0 - bbox.XMin
offset_y = 5.0 - bbox.YMin
offset_z = plate_thickness  # on top face

txt_obj.Placement.Base = App.Vector(offset_x, offset_y, offset_z)
doc.recompute()

# Extrude text downward for engraving
extrude = doc.addObject("Part::Extrusion", "TextExtrude")
extrude.Base       = txt_obj
extrude.Dir        = (0, 0, -1)
extrude.LengthFwd  = engraving_depth
extrude.Solid      = True
extrude.Reversed   = False
doc.recompute()

# Cut engraving from plate
engraved_shape = plate_with_holes.cut(extrude.Shape)

# FINALIZE MODEL
plate_obj = doc.addObject("Part::Feature", "SupportCable")
plate_obj.Label = sanitized_title
plate_obj.Shape = engraved_shape
doc.recompute()

# EXPORTS
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([plate_obj], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(engraved_shape, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    engraved_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)


# ------------------------ OBJ export with separate meshes ------------------------
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [plate_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)
