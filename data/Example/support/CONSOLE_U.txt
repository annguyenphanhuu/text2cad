#Example I would like a reinforced U-shaped bracket made of S235 steel sheet, with a thickness of 4 mm.
# Base: 150×550 mm
# Left flange: 150×550 mm
# Right flange: 150×550 mm
# It should have two vertical flanges bent at 120°, and a flat base.
# Left flange: 2 rectangular holes 10×20 mm, spaced 100 mm apart (center-to-center), centered on the flange length.
# The holes are oriented with their long side along the flange length, and their centers are aligned along the flange length.
# The hole centers are located 20 mm from the bend line (i.e., 20 mm along the flange height direction from the bend).
# Left flange: 2 oblong holes 10×15 mm, spaced 30 mm apart (center-to-center along the flange height direction).
# Long axis of each oblong along the flange length direction. Centered in length. One hole 15 mm from the outer edge.
# Base: 4 oblong holes 8×15 mm in a 2×2 grid, spaced 300 mm along the length direction.
# Long axis along the length direction. Front row 20 mm from front face.
# Width-direction positions: first hole 25 mm from the left bend, second hole 80 mm center-to-center from the first.
# Right flange: 2 oblong holes 25×50 mm, nearest oblong edge 30 mm from corner edge (edge-to-edge).
# Long axis along the flange length direction. One hole near the front corner, one near the rear corner.
# Bending radius: 4 mm.
# Shape type: U-shaped
# ADDED: 2 small return wings (15 mm wide, bent at 90°) on top of left and right flanges.
# Shape type: U-shaped

import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *


# Define the sanitized_title variable  
sanitized_title = "Console_U_S235_550x200x100x4_Wings"   # DO NOT CHANGE THIS VALUE

# Document Setup
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# Parameters (mm)
dim_x          = 150.0   
dim_y          = 550.0   
thickness      =   4.0  
flange_height  = 150.0   # Height of vertical flanges
bend_angle_deg_left  = 90.0  # Bend angle for LEFT flange
bend_angle_deg_right = 90.0  # Bend angle for RIGHT flange
bend_radius    =   thickness   # Default inner bend radius

# Wing (return flange) parameters
wing_length    =  15.0   # Small return wing width (bent outward from top of each flange)
wing_angle_deg =  90.0   # Wing bend angle

# Part.makeUshape() return:
# Base: dim_x×dim_y
# Left flange: flange_height_left×dim_y
# Right flange: flange_height_right×dim_y

# 1. Create U-shaped console
u_shape = Part.makeUShape(
    dim_x=dim_x,        
    dim_y=dim_y,        
    thickness=thickness,
    flange_height_left=flange_height,    # Left vertical
    flange_height_right=flange_height,   # Right vertical
    bend_angle_deg_left=bend_angle_deg_left,    
    bend_angle_deg_right=bend_angle_deg_right,  
    bend_radius=bend_radius
)


# 3 & 4. Add all features on LEFT flange:
# dim_y = 550 > flange_height = 150, so span length is dim_y and span width is flange_height

# ── Tool A: 10×20 mm rectangular holes ───────────────────────────────────────
rect_size_y  = 20.0              # long axis along span length
rect_size_z  = 10.0              # short axis along span width
rect_depth_x = thickness + 2.0  # cut depth through flange

# Y-centers: 2 holes spaced 100 mm apart, centered in length 
rect_spacing_y = 100.0
rect_y1 = (dim_y - rect_spacing_y) / 2.0   # = 225.0 mm
rect_y2 = rect_y1 + rect_spacing_y          # = 325.0 mm

# Z-center: 20 mm from the bend line
rect_center_z = 20.0

# Two separate instances (batch function modifies Placement in-place)
rect_tool_1 = Part.makeBox(
    rect_size_z,                                           # param1: Z extent
    rect_size_y,                                           # param2: Y extent
    rect_depth_x,                                          # param3: cut depth along X
    App.Vector(0, rect_y1 + rect_size_y / 2.0, rect_center_z - rect_size_z / 2.0),  # pnt = (0, cy+sy/2, cz-sz/2)
    App.Vector(1, 0, 0)
)
rect_tool_2 = Part.makeBox(
    rect_size_z,
    rect_size_y,
    rect_depth_x,
    App.Vector(0, rect_y2 + rect_size_y / 2.0, rect_center_z - rect_size_z / 2.0),
    App.Vector(1, 0, 0)
)

# ── Tool B: 10×15 mm oblong holes ────────────────────────────────────────────
oblong_size_z  = 10.0              # short axis along span width
oblong_size_y  = 15.0              # long axis along span length 
oblong_depth_x = thickness + 2.0   # cut depth through flange

# Y-center: centered in length
oblong_center_y = dim_y / 2.0      # = 275.0 mm

# Z-centers: 2 holes aligned along span width, 30 mm apart
oblong_z_A = flange_height - 15.0  # = 135.0 mm (15 mm from outer edge)
oblong_z_B = oblong_z_A - 30.0     # = 105.0 mm

# Two separate instances
oblong_tool_1 = Part.makeOblong(
    oblong_size_z,                                                                    # param1: Z extent
    oblong_size_y,                                                                    # param2: Y extent
    oblong_depth_x,                                                                   # param3: cut depth along X
    App.Vector(0, oblong_center_y + oblong_size_y / 2.0, oblong_z_A - oblong_size_z / 2.0),  # pnt = (0, cy+sy/2, cz-sz/2)
    App.Vector(1, 0, 0)
)
oblong_tool_2 = Part.makeOblong(
    oblong_size_z,
    oblong_size_y,
    oblong_depth_x,
    App.Vector(0, oblong_center_y + oblong_size_y / 2.0, oblong_z_B - oblong_size_z / 2.0),
    App.Vector(1, 0, 0)
)

# ── Batch-cut all left flange features in a single call ──────────────────────
left_flange_tools = [
    {"tool_shape": rect_tool_1,   "hole_center_y": rect_y1,         "hole_center_z": rect_center_z},
    {"tool_shape": rect_tool_2,   "hole_center_y": rect_y2,         "hole_center_z": rect_center_z},
    {"tool_shape": oblong_tool_1, "hole_center_y": oblong_center_y, "hole_center_z": oblong_z_A},
    {"tool_shape": oblong_tool_2, "hole_center_y": oblong_center_y, "hole_center_z": oblong_z_B},
]

u_shape = map_and_cut_leg2_batch(
    bracket_shape=u_shape,
    tools=left_flange_tools,
    angle_deg=bend_angle_deg_left,
    dim_y=dim_y,
    thickness=thickness,
    flange_height=flange_height,
    bend_radius=bend_radius
)

# 5. Add 4 oblong holes 8×15 mm (2×2 grid) on BASE
# ─────────────────────────────────────────────────────────────────
# Specification:
#   Size: 8 mm (D = minor diameter, short axis along X = base width direction)
#          × 15 mm (L = total slot length, long axis along Y = base length direction)
#   Count: 4 holes in a 2×2 grid
#   Long axis: along Y (length direction)
#   Grid spacing along Y (length): 300 mm center-to-center
#   Y-positions: Y1 = 20 mm (20 mm from front face Y=0), Y2 = 20 + 300 = 320 mm
#   X-positions: 80 mm center-to-center spacing along X, starting from left bend
#     X1 = 25 mm (25 mm from left bend at X=0)
#     X2 = X1 + 80 = 25 + 80 = 105 mm

base_oblong_diameter      = 8.0             # minor diameter (short axis, fillet = D/2 = 4 mm)
base_oblong_length        = 15.0            # total slot length (long axis along Y)
base_oblong_depth  = thickness + 2.0 # cut depth through base = 6.0 mm

# Y-positions: front row 20 mm from front face (Y=0), rear row 300 mm further
base_oblong_spacing_y = 300.0
base_oblong_y1 = 20.0                                    # = 20 mm from front face Y=0
base_oblong_y2 = base_oblong_y1 + base_oblong_spacing_y  # = 320 mm

# X-positions: 80 mm center-to-center spacing along X, starting from X1=25 mm
base_oblong_spacing_x = 80.0
base_oblong_x1 = 25.0                                    # = 25 mm from left bend line (X=0)
base_oblong_x2 = base_oblong_x1 + base_oblong_spacing_x  # = 25 + 80 = 105 mm

# Build 4 oblong tools and cut as a compound
base_oblong_tools = []
for cx in [base_oblong_x1, base_oblong_x2]:
    for cy in [base_oblong_y1, base_oblong_y2]:
        # pnt = lower-left corner: (center_x - D/2, center_y - L/2, 0.0)
        # dir=(0,0,1): no ADD compensation — standard subtract-half for both axes
        tool = Part.makeOblong(
            base_oblong_diameter,                        # param1: D (short axis, fillet along X)
            base_oblong_length,                          # param2: L (long axis along Y)
            base_oblong_depth,                           # param3: cut depth in +Z
            App.Vector(
                cx,               
                cy,                 
                0.0                                   
            ),
            App.Vector(0, 0, 1)
        )
        base_oblong_tools.append(tool)

base_oblong_compound = Part.makeCompound(base_oblong_tools)
u_shape = u_shape.cut(base_oblong_compound)

# ── Feature [D]: 2 oblong holes Ø25×50 mm on RIGHT flange ────────────────────
# dim_y = 550 > flange_height = 150, so span length is dim_y and span width is flange_height

# ─────────────────────────────────────────────────────────────────────────────
right_oblong_D     = 25.0              # minor diameter  — short axis along Z
right_oblong_L     = 50.0             # total slot length — long axis along Y
right_oblong_depth = thickness + 2.0  # cut depth through right flange wall

# Z-center: edge-to-edge 30 mm from top/outer edge
right_oblong_center_z = flange_height - 30.0 - right_oblong_D / 2.0  # = 107.5 mm

# Y-centers: nearest oblong edge 30 mm from each end corner
right_oblong_center_y1 = 30.0 + right_oblong_L / 2.0           # = 55.0 mm  (near Y=0)
right_oblong_center_y2 = dim_y - 30.0 - right_oblong_L / 2.0  # = 495.0 mm (near Y=dim_y)

# Two separate instances (Placement is modified in-place by batch function)
right_oblong_tool_1 = Part.makeOblong(
    right_oblong_D,                                                        # param1: Z extent (short axis)
    right_oblong_L,                                                        # param2: Y extent (long axis)
    right_oblong_depth,                                                    # param3: cut depth
    App.Vector(
        dim_x - thickness,                                                 # pnt.x = dim_x - thickness (90° right wall inner face)
        right_oblong_center_y1 + right_oblong_L / 2.0,                    # pnt.y = center_y + L/2  (ADD)
        right_oblong_center_z - right_oblong_D / 2.0                      # pnt.z = center_z - D/2  (SUBTRACT)
    ),
    App.Vector(1, 0, 0)    # canonical cut direction
)
right_oblong_tool_2 = Part.makeOblong(
    right_oblong_D,
    right_oblong_L,
    right_oblong_depth,
    App.Vector(
        dim_x - thickness,                                                 # pnt.x = dim_x - thickness (90° right wall)
        right_oblong_center_y2 + right_oblong_L / 2.0,                    # pnt.y = center_y + L/2  (ADD)
        right_oblong_center_z - right_oblong_D / 2.0                      # pnt.z = center_z - D/2  (SUBTRACT)
    ),
    App.Vector(1, 0, 0)    # canonical cut direction
)

right_flange_tools = [
    {"tool_shape": right_oblong_tool_1, "hole_center_y": right_oblong_center_y1, "hole_center_z": right_oblong_center_z},
    {"tool_shape": right_oblong_tool_2, "hole_center_y": right_oblong_center_y2, "hole_center_z": right_oblong_center_z},
]

u_shape = map_and_cut_right_flange_batch(
    bracket_shape=u_shape,
    tools=right_flange_tools,
    angle_deg=bend_angle_deg_right,
    dim_x=dim_x,
    dim_y=dim_y,
    thickness=thickness,
    flange_height=flange_height,
    bend_radius=bend_radius
)

# ── Register u_shape as a Part::Feature so SMBendWall can reference it ──────
u_shape_obj = doc.addObject("Part::Feature", "UShape_Base")
u_shape_obj.Label = "UShape_Base"
u_shape_obj.Shape = u_shape
doc.recompute()

# ============================================================
# 6. ADD 2 SMALL RETURN WINGS (top of left & right flanges)
# ============================================================
u_shape_obj = AddUShapeLeftReturn(
    shape=u_shape_obj,
    thickness=thickness,
    dim_x=dim_x,
    dim_y=dim_y,
    flange_height=flange_height,
    wing_length=wing_length,
    wing_angle_deg=wing_angle_deg,
    bend_radius=bend_radius,
    flange_angle_deg=bend_angle_deg_left,
    wing_type="inside",
)

u_shape_obj = AddUShapeRightReturn(
    shape=u_shape_obj,
    thickness=thickness,
    dim_x=dim_x,
    dim_y=dim_y,
    flange_height=flange_height,
    wing_length=wing_length,
    wing_angle_deg=wing_angle_deg,
    bend_radius=bend_radius,
    flange_angle_deg=bend_angle_deg_right,
    wing_type="inside",
)

# ── Final object after all wings ─────────────────────────────────────────────
final_obj = u_shape_obj
final_obj.Label = sanitized_title
doc.recompute()

# Export directory handling
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([final_obj], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

final_shape = final_obj.Shape

# Analyze final shape geometry
geometry_result = analyze_geometry(final_shape, material="{mapped_material}")

# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    final_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)


# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [final_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape            = face,
            LinearDeflection = 0.1,
            AngularDeflection= 0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh  = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)

# Final recompute (optional)
doc.recompute()
