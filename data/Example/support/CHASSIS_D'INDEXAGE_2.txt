#Example a guiding base with positioning pins featuring multiple notches. A 400 × 400 × 5 mm plate is required, with 12 Ø6.2 holes arranged in two rows of six, spaced 150 mm apart and centered. The notches should be spaced every 40 mm, giving a total pitch of 200 mm between the two ends. Add Ø6 × 20 mm pins in each hole, protruding 20 mm on one side. Also, add four M8 holes in the corners, each located 40 mm from the edges.
# Shape type: plate

# -*- coding: utf-8 -*-
import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# Document setup
sanitized_title = "Guiding_Base_400x400x5_with_Pins_and_Holes"
doc_name = sanitized_title
if App.ActiveDocument is None or App.ActiveDocument.Name != doc_name:
    doc = App.newDocument(doc_name)
else:
    doc = App.ActiveDocument

# Parameters
plate_length = 400.0  # X
plate_width  = 400.0  # Y
plate_thick  = 5.0    # Z

# 1. Create base plate
plate_base = Part.makeBox(plate_length, plate_width, plate_thick, App.Vector(0, 0, 0))

# 2. Create D6.2 holes (12 holes)
hole_d6p2_radius = 6.2 / 2.0
# Spacing for 6 holes across the X direction (two rows)
margin_x = 50.0
x_spacing = (plate_length - 2.0 * margin_x) / (6 - 1)  # 62.5 mm
x_positions = [margin_x + i * x_spacing for i in range(6)]

# Y positions for two rows
y_positions = [100.0, 300.0]

holes_d62 = []
for y in y_positions:
    for x in x_positions:
        cyl = Part.makeCylinder(
            hole_d6p2_radius,
            plate_thick + 2.0,  # extend slightly beyond thickness
            App.Vector(x, y, plate_thick),
            App.Vector(0, 0, -1)
        )
        holes_d62.append(cyl)

holes_d62_compound = Part.makeCompound(holes_d62)
plate_after_d62 = plate_base.cut(holes_d62_compound)

# 3. Create M8 corner holes (4 holes, use compound since count >= 4)
hole_m8_radius   = 8.0 / 2.0
# Corner M8 holes - 40 mm from edges
corner_offset = 40.0
corner_positions = [
    (corner_offset, corner_offset),
    (plate_length - corner_offset, corner_offset),
    (corner_offset, plate_width - corner_offset),
    (plate_length - corner_offset, plate_width - corner_offset),
]

corner_holes = []
for (x, y) in corner_positions:
    cyl = Part.makeCylinder(
        hole_m8_radius,
        plate_thick + 2.0,
        App.Vector(x, y, plate_thick),
        App.Vector(0, 0, -1)
    )
    corner_holes.append(cyl)

corner_holes_compound = Part.makeCompound(corner_holes)
plate_after_m8 = plate_after_d62.cut(corner_holes_compound)

# 4. Create guiding pins (D6 x 20 mm) and fuse to plate
pin_radius       = 6.0 / 2.0
pin_length       = 20.0

pins = []
for y in y_positions:
    for x in x_positions:
        pin = Part.makeCylinder(
            pin_radius,
            pin_length,
            App.Vector(x, y, plate_thick),
            App.Vector(0, 0, 1)
        )
        pins.append(pin)

pins_compound = Part.makeCompound(pins)
guiding_base_complete = plate_after_m8.fuse(pins_compound)

# Add to document
part_obj = doc.addObject("Part::Feature", "Guiding_Base_Complete")
part_obj.Shape = guiding_base_complete
doc.recompute()

# Create output directory
output_dir_abs = f"/app/storage/{sanitized_title}/output"
if not os.path.exists(output_dir_abs):
    os.makedirs(output_dir_abs)

# Export to STEP
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([part_obj], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(guiding_base_complete, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    guiding_base_complete,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)

# Export OBJ with separate meshes for each face
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [part_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(Shape=face, LinearDeflection=0.1, AngularDeflection=0.523599)
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)
