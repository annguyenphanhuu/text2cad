#Example bending choose edge
# This example demonstrates how to use the SheetMetal workbench to bend a plate along a specific edge.
# It includes a function to find an edge by its coordinates.
# -*- coding: utf-8 -*-
#
import sys
import os
import FreeCAD as App
import Part

# 2. SheetMetal workbench bootstrap
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()

# Find sheetmetal directory - search upward from script location
current_dir = script_dir
sheetmetal_path = None
for _ in range(5):  # Search up to 5 levels up
    test_path = os.path.join(current_dir, "sheetmetal")
    if os.path.exists(test_path) and os.path.exists(os.path.join(test_path, "SheetMetalCmd.py")):
        sheetmetal_path = test_path
        break
    current_dir = os.path.dirname(current_dir)

if sheetmetal_path is None:
    raise ImportError("SheetMetal workbench not found")

if sheetmetal_path not in sys.path:
    sys.path.append(sheetmetal_path)

# Import SheetMetal
import SheetMetalCmd
import SheetMetalTools

# Add missing function to avoid errors
def taskRestoreDefaults(obj, default_vars):
    pass

SheetMetalTools.taskRestoreDefaults = taskRestoreDefaults

# Create document
doc = App.newDocument("SMBendWallSimplified")
thickness = 5.0
dim_x = 100.0
wall_length = 200.0
dim_y = 50.0
angle_deg = 90.0
bend_radius = thickness # Default bend radius


# Create base box
base_box = Part.makeBox(dim_x, dim_y, thickness)   
base_obj = doc.addObject("Part::Feature", "BaseBox")
base_obj.Shape = base_box
base_obj.Label = "Base Box"

doc.recompute()

# Function to find edge by coordinates
def find_edge_by_coordinates(obj, start_point, end_point, tolerance=1e-6):
    """Find edge that matches start and end coordinates"""

    for i, edge in enumerate(obj.Shape.Edges):
        edge_start = edge.Vertexes[0].Point
        edge_end = edge.Vertexes[1].Point

        # Check if edge matches start->end or end->start
        if ((edge_start.distanceToPoint(start_point) < tolerance and
             edge_end.distanceToPoint(end_point) < tolerance) or
            (edge_start.distanceToPoint(end_point) < tolerance and
             edge_end.distanceToPoint(start_point) < tolerance)):
            return f"Edge{i+1}"
    return None

# Create SMBendWall
wall_obj = doc.addObject("Part::FeaturePython", "SMBendWall")

# Define target edge coordinates
start_point = App.Vector(0, 0, 0)
end_point = App.Vector(100, 0, 0)

# Find the edge that matches these coordinates
target_edge = find_edge_by_coordinates(base_obj, start_point, end_point)

if target_edge:
    wall_feature = SheetMetalCmd.SMBendWall(wall_obj, base_obj, [target_edge])
else:

# Set properties
wall_obj.length = wall_length
# Remember angle is 180 - angle_deg
wall_obj.angle = 180 - angle_deg
wall_obj.radius = bend_radius

# Recompute
doc.recompute()

# Export to STEP
step_path = os.path.join(os.getcwd(), "SMBendWallSimplified.step")
Part.export([base_obj, wall_obj], step_path)
