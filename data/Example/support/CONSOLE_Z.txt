#Example I would like a reinforced Z-shaped bracket made of S235 steel sheet, with a thickness of 4 mm, with 3mm corner fillets.
# The overall dimensions are: length 550 mm, web height 150 mm, top flange height 100 mm, bottom flange height 100 mm.
# It should have a top flange and a bottom flange bent at 90°, connected by a central vertical web.
# Top flange: 2 rectangular holes 10×20 mm, spaced 100 mm apart (center-to-center), centered on the flange length.
# The holes are oriented with their long side along the flange length, and their centers are aligned along the flange length.
# The hole centers are located 20 mm from the bend line.
# Top flange: 2 oblong holes 10×15 mm, spaced 30 mm apart (center-to-center along the flange height).
# Long axis of each oblong along the flange length. Centered in length. One hole 15 mm from the outer edge.
# Central web: 4 oblong holes 8×15 mm in a 2×2 grid, spaced 300 mm along the length.
# Long axis along the length direction. Front row 20 mm from the front face.
# Across the web height, the holes are 25 mm from the top bend line, with an 80 mm center-to-center spacing.
# Bottom flange: 2 round holes (10 mm diameter), spaced 500 mm apart, centered in length.
# The holes are located 15 mm from the outer edge/bottom edge.
# Bottom flange: 2 oblong holes 20x14 mm, spaced 100 mm apart, centered in length.
# Long axis along the flange width. Edge distance = 20 mm from the outer/bottom edge
# (mep lo → mep ngoai, KHÔNG phai tu tam lo; outer edge = Z = -(bottom_flange_height - thickness)).
# Bottom flange: rectangular corner notch 30×20 mm, long axis along width, 50 mm from corner in BOTH Y and Z.
# ⚠️ CORNER NOTCH rule: "Xmm from corner" applies to EACH axis independently:
#   Y-axis: near-edge of notch at Y=X  → pnt_y = offset_y + size_y  (ADD full size, dir=(1,0,0) goes -Y)
#   Z-axis: near-edge of notch at outer_edge_z + X  → pnt_z = outer_edge_z + offset_z  (NOT flush at outer edge)
# Bending radius: 4 mm.
# Shape type: Z-shaped

import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# Define the sanitized_title variable
sanitized_title = "Console_Z_S235_550x150x100x100x4"   # DO NOT CHANGE THIS VALUE

# Document Setup
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# Parameters (mm)
# Sheet bent along the 550mm length (dim_y = 550mm)
dim_x              = 150.0   # Web height (X dimension — cross-section span between the two bend lines)
dim_y              = 550.0   # Extrusion depth (Y dimension). IMPORTANT: ALWAYS MATCHING LENGTH TO dim_y
thickness          =   4.0   # Sheet thickness
top_flange_height  = 100.0   # Height of top flange (at X=0 side, bends along Z = web_height)
bottom_flange_height = 100.0 # Height of bottom flange (at X=dim_x side, bends along Z = 0)
top_bend_angle_deg   = 120.0   # Top bend angle   
bottom_bend_angle_deg = 150.0  # Bottom bend angle
bend_radius        =   thickness   # Default inner bend radius

# 1. Create Z-shaped console
z_shape = Part.makeZShape(
    dim_x=dim_x,                          # Web height (X dimension). IMPORTANT: ALWAYS MATCHING WEB TO dim_x
    dim_y=dim_y,                          # Extrusion depth (Y dimension). IMPORTANT: ALWAYS MATCHING LENGTH TO dim_y
    thickness=thickness,
    top_flange_height=top_flange_height,   # Top flange height/hauteur (Z dimension, at X=0 side)
    bottom_flange_height=bottom_flange_height, # Bottom flange height/hauteur (Z dimension, at X=dim_x side)
    top_bend_angle_deg=top_bend_angle_deg,
    bottom_bend_angle_deg=bottom_bend_angle_deg,
    bend_radius=bend_radius
)

# Apply 3mm corner fillets on all thickness edges
fillet_edges = []
for edge in z_shape.Edges:
    if abs(edge.Length - thickness) < 1e-6:
        fillet_edges.append(edge)
if fillet_edges:
    z_shape = z_shape.makeFillet(3.0, fillet_edges)


# 3. Add 2 rectangular holes 10×20 mm on TOP flange
# ─────────────────────────────────────────────────────────────────
# Specification:
#   Size: 10 mm (short, along Z = flange height) × 20 mm (long, along Y = flange length)
#   Count: 2 holes
#   Spacing: 100 mm center-to-center along Y (flange length direction)
#   Position along Y: centered in dim_y=550 → Y1=225 mm, Y2=325 mm
#   Position along Z: 20 mm from the bend line (Z=0 at bend, same as U left flange) → center_z = 20.0 mm
#
# Part.makeBox(size_z, size_y, depth_x, pnt, dir=(1,0,0)) mapping on TOP flange:
#   param1 (size_z=10) → +Z from pnt.z → pnt.z = center_z − size_z/2  (SUBTRACT)
#   param2 (size_y=20) → −Y from pnt.y → pnt.y = center_y + size_y/2  (ADD)
#   param3 (depth_x)   → +X = cut depth through flange
#   pnt.x = 0 → start at outer face of top flange (template: Z Top Flange pnt.x = 0)

rect_size_y  = 20.0             # long axis along Y (flange length direction)
rect_size_z  = 10.0             # short axis along Z (flange height direction)
rect_depth_x = thickness + 2.0 # cut depth through flange = 6.0 mm

# Y-centers: 2 holes spaced 100 mm apart, centered in dim_y=550
rect_spacing_y = 100.0
rect_y1 = (dim_y - rect_spacing_y) / 2.0   # = (550 - 100) / 2 = 225.0 mm
rect_y2 = rect_y1 + rect_spacing_y          # = 325.0 mm

# Z-center: 20 mm from the top bend line (Z=0 at bend, same coordinate space as U left flange)
rect_center_z = 20.0  # = 20.0 mm from bend (Z=0)

# Build 2 tool boxes
top_flange_tools = []
for cy in [rect_y1, rect_y2]:
    # pnt.x = 0  → start at outer face of top flange (Z Top Flange template: pnt.x = 0.0)
    # pnt.y = cy + rect_size_y/2  → Y extent occupies [cy−size_y/2, cy+size_y/2] in −Y direction
    # pnt.z = center_z − rect_size_z/2  → Z extent occupies [center_z−size_z/2, center_z+size_z/2] in +Z direction
    top_flange_tools.append({
        "tool_shape": Part.makeBox(
            rect_size_z,                                    # param1: Z extent → +Z from pnt.z (SUBTRACT ½)
            rect_size_y,                                    # param2: Y extent → −Y from pnt.y (ADD ½)
            rect_depth_x,                                   # param3: cut depth → +X from pnt.x
            App.Vector(0, cy + rect_size_y / 2.0, rect_center_z - rect_size_z / 2.0),
            App.Vector(1, 0, 0)
        ),
        "hole_center_y": cy,
        "hole_center_z": rect_center_z,
    })

# 4. Add 2 oblong holes 10×15 mm on TOP flange
# ─────────────────────────────────────────────────────────────────
# Specification:
#   Size: 10 mm (short, along Z = height) × 15 mm (long, along Y = length)
#   Count: 2 holes, aligned along Z (flange height direction)
#   Long axis: along Y (flange length direction)
#   Spacing: 30 mm center-to-center along Z
#   Center along Y: centered in dim_y=550 → center_y = 275.0 mm
#   Z positions (Z=0 at bend, Z=top_flange_height at outer edge — same as U left flange):
#     Hole A (near outer edge): center_z_A = top_flange_height − 15 = 85 mm  (15 mm from outer edge Z=100)
#     Hole B (toward bend):     center_z_B = center_z_A − 30 = 55 mm
#
# Part.makeOblong(param1, param2, param3, pnt, dir=(1,0,0)) on top flange:
#   param1 (size_z=10) → +Z from pnt.z → pnt.z = center_z − param1/2  (SUBTRACT)
#   param2 (size_y=15) → −Y from pnt.y → pnt.y = center_y + param2/2  (ADD)
#   param3 (depth_x)   → +X = cut depth through flange
#   pnt.x = 0 → start at outer face of top flange (template: Z Top Flange pnt.x = 0)
#   ⚠️ Long axis along Y → param1 = short(Z) = 10, param2 = long(Y) = 15

oblong_size_z  = 10.0             # short axis along Z (flange height direction)
oblong_size_y  = 15.0             # long axis along Y (flange length direction)
oblong_depth_x = thickness + 2.0  # cut depth through flange = 6.0 mm

# Y-center: centered in dim_y=550
oblong_center_y = dim_y / 2.0     # = 275.0 mm

# Z-centers: 2 holes aligned along Z, 30 mm apart
#   (Z=0 at bend, Z=top_flange_height at outer edge — same coordinate space as U left flange)
oblong_z_A = top_flange_height - 15.0  # = 100 − 15 = 85.0 mm (15 mm from outer edge Z=100)
oblong_z_B = oblong_z_A - 30.0         # = 55.0 mm

# pnt.y is the same for both holes (same center_y)
oblong_pnt_y = oblong_center_y + oblong_size_y / 2.0  # = 275 + 7.5 = 282.5 mm

# Build 2 oblong tools
for cz in [oblong_z_A, oblong_z_B]:
    # pnt.z = center_z − oblong_size_z/2  (SUBTRACT because param1 goes in +Z)
    top_flange_tools.append({
        "tool_shape": Part.makeOblong(
            oblong_size_z,                              # param1: Z extent → +Z from pnt.z (SUBTRACT ½)
            oblong_size_y,                              # param2: Y extent → −Y from pnt.y (ADD ½)
            oblong_depth_x,                             # param3: cut depth → +X from pnt.x
            App.Vector(0, oblong_pnt_y, cz - oblong_size_z / 2.0),
            App.Vector(1, 0, 0)
        ),
        "hole_center_y": oblong_center_y,
        "hole_center_z": cz,
    })

# Cut all top flange tools via map_and_cut_leg2_batch (handles any bend angle)
z_shape = map_and_cut_leg2_batch(
    z_shape,
    top_flange_tools,
    angle_deg=top_bend_angle_deg,
    dim_y=dim_y,
    thickness=thickness,
    flange_height=top_flange_height,
    bend_radius=bend_radius,
)

# 5. Add 4 oblong holes 8×15 mm (2×2 grid) on CENTRAL WEB
# ─────────────────────────────────────────────────────────────────
# Specification:
#   Size: 8 mm (D = minor diameter, short axis along X = web width direction)
#          × 15 mm (L = total slot length, long axis along Y = web length direction)
#   Count: 4 holes in a 2×2 grid
#   Long axis: along Y (length direction)
#   Grid spacing along Y (length): 300 mm center-to-center
#   Y-positions: Y1 = 20 mm (20 mm from front face Y=0), Y2 = 20 + 300 = 320 mm
#   X-positions: 80 mm center-to-center spacing along X, starting from left bend (X=0)
#     X1 = 25 mm (25 mm from top bend line at X=0)
#     X2 = X1 + 80 = 25 + 80 = 105 mm
#
# Part.makeOblong(D, L, cut_depth, pnt, dir=(0,0,1)) on CENTRAL WEB — same as BASE template:
#   param1 = D = 8  → minor diameter (short axis, fillet radius = D/2 = 4)
#   param2 = L = 15 → total slot length (long axis along Y)
#   param3 = cut_depth_z = thickness + 2.0
#   pnt = App.Vector(center_x - D/2, center_y - L/2, 0.0)
#   dir=(0,0,1): no coordinate compensation — pnt is lower-left corner (subtract-half for both X and Y).
#   pnt.z = 0.0 → start at outer (bottom) face of central web (same as Base template: pnt.z = 0.0)

web_oblong_diameter      = 8.0             # minor diameter (short axis, fillet = D/2 = 4 mm)
web_oblong_length        = 15.0            # total slot length (long axis along Y)
web_oblong_depth  = thickness + 2.0 # cut depth through web = 6.0 mm

# Y-positions: front row 20 mm from front face (Y=0), rear row 300 mm further
web_oblong_spacing_y = 300.0
web_oblong_y1 = 20.0                                   # = 20 mm from front face Y=0
web_oblong_y2 = web_oblong_y1 + web_oblong_spacing_y   # = 320 mm

# X-positions: 80 mm center-to-center spacing along X, starting from X1=25 mm (from top bend at X=0)
web_oblong_spacing_x = 80.0
web_oblong_x1 = 25.0                                   # = 25 mm from top bend line (X=0)
web_oblong_x2 = web_oblong_x1 + web_oblong_spacing_x   # = 25 + 80 = 105 mm

# Build 4 oblong tools and cut as a compound
web_oblong_tools = []
for cx in [web_oblong_x1, web_oblong_x2]:
    for cy in [web_oblong_y1, web_oblong_y2]:
        # pnt = lower-left corner: (center_x - D/2, center_y - L/2, 0.0)
        # dir=(0,0,1): no ADD compensation — standard subtract-half for both axes
        tool = Part.makeOblong(
            web_oblong_diameter,                        # param1: D (short axis, fillet along X)
            web_oblong_length,                          # param2: L (long axis along Y)
            web_oblong_depth,                           # param3: cut depth in +Z
            App.Vector(
                cx - web_oblong_diameter / 2.0,         # pnt.x = center_x - D/2
                cy - web_oblong_length / 2.0,           # pnt.y = center_y - L/2
                0.0                                     # pnt.z = 0.0 (outer/bottom face of central web)
            ),
            App.Vector(0, 0, 1)
        )
        web_oblong_tools.append(tool)

web_oblong_compound = Part.makeCompound(web_oblong_tools)
z_shape = z_shape.cut(web_oblong_compound)

# 6. Bottom flange features — ALL via map_and_cut_bottom_flange_batch
# ─────────────────────────────────────────────────────────────────
# All bottom flange cutting operations are grouped here and processed
# via a single batch call that handles any bend angle automatically.
#
# COORDINATE CONVENTION for bottom flange (90° natural-coordinate, same as right flange):
#   • hole_center_y ∈ [0, dim_y]                  — position along extrusion length
#   • hole_center_z ∈ [0, bottom_flange_height]    — distance from bend line along flange surface
#   • pnt.x = dim_x - thickness  (inner face of 90° bottom wall — matrix handles placement)
#   • dir = (1, 0, 0)
#   • The batch function handles rotation / translation to the actual angled face.
#
# ── 6a. 2 round holes (D=10) ────────────────────────────────────
# Specification:
#   Size: circular holes, diameter = 10 mm (radius = 5.0 mm)
#   Count: 2 holes
#   Position along Y: centered in length (dim_y = 550), separated by 500 mm
#   Position along Z: 15 mm from outer edge → center_z = bottom_flange_height - 15.0

bottom_hole_radius = 5.0
bottom_spacing_y = 500.0
bottom_center_y  = dim_y / 2.0
bottom_y1 = bottom_center_y - bottom_spacing_y / 2.0
bottom_y2 = bottom_center_y + bottom_spacing_y / 2.0
bottom_z  = bottom_flange_height - 15.0   # 15 mm from outer edge

bottom_tools = []

# Round hole 1
bottom_tools.append({
    "tool_shape": Part.makeCylinder(
        bottom_hole_radius, thickness + 2.0,
        App.Vector(dim_x - thickness, bottom_y1, bottom_z), App.Vector(1, 0, 0)
    ),
    "hole_center_y": bottom_y1,
    "hole_center_z": bottom_z,
})
# Round hole 2
bottom_tools.append({
    "tool_shape": Part.makeCylinder(
        bottom_hole_radius, thickness + 2.0,
        App.Vector(dim_x - thickness, bottom_y2, bottom_z), App.Vector(1, 0, 0)
    ),
    "hole_center_y": bottom_y2,
    "hole_center_z": bottom_z,
})

# ── 6b. 2 oblong holes 20×14 mm ─────────────────────────────────
# Specification:
#   Size: 20 mm (long axis along Z = width direction) × 14 mm (short axis along Y = length direction)
#   Count: 2 holes, spaced 100 mm center-to-center along Y (length)
#   Centered in length (dim_y = 550) → center_y = 275.0 mm
#   Edge distance = 20 mm from the outer/bottom edge (mép lỗ → mép ngoài)
#   → center_z = bottom_flange_height - 20.0 - 20.0/2 = bottom_flange_height - 30.0
#   BUT using the natural-coordinate system:
#     outer_edge_z_natural = bottom_flange_height (from the bend line)
#     edge_dist = 20.0 (edge distance, not center)
#     near_edge_z = outer_edge_z_natural - edge_dist = 100 - 20 = 80.0
#     center_z = near_edge_z - size_z/2 = 80 - 10 = 70.0

bottom_oblong_size_z = 20.0    # long axis along Z (flange width)
bottom_oblong_size_y = 14.0    # short axis along Y (flange length)
bottom_oblong_depth  = thickness + 2.0

bottom_oblong_spacing_y = 100.0
bottom_oblong_center_y  = dim_y / 2.0
bottom_oblong_y1 = bottom_oblong_center_y - bottom_oblong_spacing_y / 2.0
bottom_oblong_y2 = bottom_oblong_center_y + bottom_oblong_spacing_y / 2.0

edge_dist                   = 20.0
bottom_oblong_near_edge_z   = bottom_flange_height - edge_dist  # = 100 - 20 = 80
bottom_oblong_center_z      = bottom_oblong_near_edge_z - bottom_oblong_size_z / 2.0  # = 80 - 10 = 70

for cy in [bottom_oblong_y1, bottom_oblong_y2]:
    bottom_tools.append({
        "tool_shape": Part.makeOblong(
            bottom_oblong_size_z,
            bottom_oblong_size_y,
            bottom_oblong_depth,
            App.Vector(dim_x - thickness, cy + bottom_oblong_size_y / 2.0, bottom_oblong_center_z - bottom_oblong_size_z / 2.0),
            App.Vector(1, 0, 0)
        ),
        "hole_center_y": cy,
        "hole_center_z": bottom_oblong_center_z,
    })

# ── 6c. Corner notch 30×20 mm ───────────────────────────────────
# Specification:
#   Size: 30mm (Z) × 20mm (Y), 50mm from corner in BOTH Y and Z
#   Using natural-coordinate convention:
#     Y-axis: near-edge of notch at Y = 50 → pnt_y = offset_y + size_y = 50 + 20 = 70
#     Z-axis: near-edge at outer_edge + offset = flange_height - 50 from bend line
#       → In natural coords: center_z_notch = bottom_flange_height - 50 - 30/2 = 35
#       → pnt_z = center_z_notch - size_z/2 = 35 - 15 = 20.0
#       But for makeBox with dir=(1,0,0): pnt_z = center_z - size_z/2

cn_size_z   = 30.0
cn_size_y   = 20.0
cn_depth    = thickness + 2.0
cn_offset_y = 50.0   # 50mm from Y=0 corner
cn_offset_z = 50.0   # 50mm from outer Z edge (tip of flange)

cn_center_z = bottom_flange_height - cn_offset_z - cn_size_z / 2.0  # = 100 - 50 - 15 = 35.0
cn_pnt_y    = cn_offset_y + cn_size_y   # = 50 + 20 = 70 (ADD for dir=(1,0,0))
cn_pnt_z    = cn_center_z - cn_size_z / 2.0  # = 35 - 15 = 20.0

bottom_tools.append({
    "tool_shape": Part.makeBox(
        cn_size_z, cn_size_y, cn_depth,
        App.Vector(dim_x - thickness, cn_pnt_y, cn_pnt_z),
        App.Vector(1, 0, 0)
    ),
    "hole_center_y": cn_offset_y + cn_size_y / 2.0,
    "hole_center_z": cn_center_z,
})

# ── Cut all bottom flange tools in one batch call ────────────────
z_shape = map_and_cut_bottom_flange_batch(
    z_shape,
    bottom_tools,
    angle_deg=bottom_bend_angle_deg,
    dim_x=dim_x,
    dim_y=dim_y,
    thickness=thickness,
    flange_height=bottom_flange_height,
    bend_radius=bend_radius,
)

final_obj = doc.addObject("Part::Feature", "Console_Z_Final")
final_obj.Label = sanitized_title
final_obj.Shape = z_shape
doc.recompute()

# Export directory handling
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([final_obj], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(z_shape, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    z_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)


# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [final_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape            = face,
            LinearDeflection = 0.1,
            AngularDeflection= 0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh  = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)

# Final recompute (optional)
doc.recompute()
