#Example Create an L-shaped bracket with a base of 100 mm, a height of 60 mm, and a width of 30 mm, made from 2 mm thick material. The part includes a 120° bend with an inner bending radius of 2 mm and an outer bending radius of 4 mm.
# The bracket features:
# Two holes of 6 mm diameter on the base, spaced 70 mm apart (center-to-center), and centered in width.
# One hole of 8 mm diameter, centered on the 60 mm vertical leg.
# Add 5 mm fillet radii to all corners.
# Four oblong slots (4x10 mm) on the base, 2x2 layout: 10 mm gap along length, 5 mm gap along width, oriented along the length, centered in width, 20 mm from the bottom (far) edge of the base.
# One oblong slot (20x6 mm) on the vertical wall, centered in width, with center 10 mm from top edge, oriented along the length.
# Shape type: L-bracket
# Operation family: center-to-center holes / single hole / oblong slots — NOT a linear spacing pattern.
#   → For THESE operations, hand Part.makeCylinder / Part.makeOblong is correct. Do NOT call
#     Part.resolveLBracketCrossBendHoles here. This example is NOT a template for a spacing pattern.
#
# ═══ REFERENCE NOTE — L-BRACKET BASE LINEAR HOLE PATTERN ═══
# The holes in THIS example are positioned by a center-to-center distance, so the rule below does
# NOT apply here. It applies only when a base hole pattern is given as a spacing E repeated from
# one edge distance D ("holes every E mm starting D mm from the edge"). Keep it in mind for such
# requests; reason with the REQUEST's own values (E, D and the leg lengths below are placeholders
# — never copy a number from this example):
#
#   STEP 1 — reference_edge?
#       D measured from the base's free edge (opposite the bend) -> reference_edge="far_edge"
#       D measured from the fold line itself                     -> reference_edge="bend_edge"
#
#   STEP 2 — did the user write a STOP phrase ("only on the base"/"uniquement sur la base",
#            "stop at X", "up to the bend", "in the middle")?
#       NO  stop phrase AND reference_edge=="far_edge" AND no explicit total count
#           -> OUTCOME A (CROSS-BEND): cross_bend=True, stop_position=None, hole_count=None.
#              The pattern fills the base and CONTINUES across the fold onto the vertical wall.
#              Silence about the wall is NOT a reason to keep the holes on the base.
#       YES stop phrase, OR reference_edge=="bend_edge", OR an explicit count N
#           -> OUTCOME B (BASE ONLY): cross_bend=False (REQUIRED — this is what keeps the holes
#              off the wall; stop_position alone does NOT stop the crossing).
#
#   STEP 3 — ALWAYS call Part.resolveLBracketCrossBendHoles(...) for that pattern. NEVER
#            hand-write a `for i in range(n)` loop: the helper applies the DFM bend clearance,
#            keeps the spacing continuous around the fold (it adds the inner bend_radius as the
#            arc length), and skips any hole that would land on the bend itself.
#
#   STEP 4 — cut what it returns: row["base_positions"] directly (dir=(0,0,1));
#            row["vertical_positions"] ONLY via map_and_cut_leg2_batch (never cut directly — the
#            bend angle may not be 90°). For OUTCOME B vertical_positions is always empty.
#
#   result = Part.resolveLBracketCrossBendHoles(
#       base_length=dim_x, flange_height=flange_height, bend_radius=bend_radius,
#       thickness=thickness, reference_edge="far_edge"|"bend_edge", edge_distance=D,
#       spacing=E, min_edge_distance=thickness, row_positions=[Y...],
#       hole_count=None_or_N, stop_position=None_or_X, cross_bend=True_or_False,
#       hole_radius=hole_diameter / 2.0,
#   )
#   hole_radius is REQUIRED: every clearance is measured to the hole EDGE, so without it the
#   last hole of a row overhangs the free edge and comes out clipped. min_edge_distance is the
#   FREE-edge clearance only (thickness, per LC_02); the bend-edge clearance is derived inside.

# -*- coding: utf-8 -*-
import sys
import os
import math
import FreeCAD as App
import Import
import Part
import Mesh
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# METADATA
sanitized_title = "SUPPORT_1"
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# Base horizontal : 100×30 mm
# Paroi verticale : 60×30 mm

# 1. Create the L-shaped bracket
dim_x = 100.0       # Base dimension X
dim_y = 30.0        # Base dimension Y / Flange dimension Y 
thickness = 2.0     # Sheet thickness
flange_height = 60.0    # Flange height / Flange dimension Z
bend_angle_deg =120.0   # 120° L-bracket
bend_radius = 2.0       # Inner bend radius (given: 2 mm)
fillet_radius_selected = 5.0     # 5 mm fillet radii to all corners

# Horizontal part: 100x30
# Vertical part: 60x30
bracket_shape = Part.makeLShape(
    dim_x=dim_x,
    dim_y=dim_y,    # Depth/Bend along side
    thickness=thickness,
    flange_height=flange_height,
    bend_angle_deg=bend_angle_deg,
    bend_radius=bend_radius
)

# Apply 5mm corner fillets to all thickness edges
fillet_edges = []
for edge in bracket_shape.Edges:
    if abs(edge.Length - thickness) < 1e-6:
        fillet_edges.append(edge)
if fillet_edges:
    bracket_shape = bracket_shape.makeFillet(fillet_radius_selected, fillet_edges)

# 2. Add two Ø6 mm holes on the base (leg1):
# Spaced 70 mm apart (center-to-center) along X, centered in width (dim_y)
leg1_hole_radius = 6.0 / 2.0   # Ø6mm -> R = 3.0mm

# 2.1 Both holes share the same Y: centered in width (dim_y)
leg1_hole_y = dim_y / 2.0      # Centered in width (Y dimension)

# 2.2 Spaced 70 mm apart along X (base length = 100 mm), centered on the base
leg1_center_to_center_distance = 70.0  # Distance between two holes along X
leg1_hole1_x = (dim_x - leg1_center_to_center_distance) / 2.0   # = 15.0 mm
leg1_hole2_x = leg1_hole1_x + leg1_center_to_center_distance     # = 85.0 mm

# Hole 1 on leg1 — cylinder cut upward through the base (dir=(0,0,1), pnt.z=0.0)
leg1_hole_cut_depth = thickness + 2.0
leg1_hole1_tool = Part.makeCylinder(
    leg1_hole_radius,
    leg1_hole_cut_depth,
    App.Vector(leg1_hole1_x, leg1_hole_y, 0.0),  # start at bottom face
    App.Vector(0, 0, 1)  # cut upward through the base plate
)
bracket_shape = bracket_shape.cut(leg1_hole1_tool)

# Hole 2 on leg1
leg1_hole2_tool = Part.makeCylinder(
    leg1_hole_radius,
    leg1_hole_cut_depth,
    App.Vector(leg1_hole2_x, leg1_hole_y, 0.0),
    App.Vector(0, 0, 1)
)
bracket_shape = bracket_shape.cut(leg1_hole2_tool)

# 3. Add one Ø8 mm hole centered on the 60 mm vertical leg (leg2):
# Centered both in the length (dim_y) and in height (flange_height)
leg2_hole_radius = 8.0 / 2.0   # Ø8mm -> R = 4.0mm

# 3.1 Centered along width (dim_y)
leg2_hole_y = dim_y / 2.0

# 3.2 Centered in the vertical leg height
leg2_hole_z = flange_height / 2.0

# Hole on leg2 - created at 90° natural coordinates:
# pnt encodes actual (y, z) target position on the face — no origin shift needed.
# map_and_cut_leg2_batch transforms this to fit the actual 120° bend angle.
cut_depth_hole = thickness + 2.0
leg2_hole_tool = Part.makeCylinder(
    leg2_hole_radius,
    cut_depth_hole,
    App.Vector(0, leg2_hole_y, leg2_hole_z),   # pnt = (0, cy, cz) — 90° convention
    App.Vector(1, 0, 0)
)

# 4. Add 4 oblong slots on the BASE (leg1) — oriented along X
# Layout: 2 columns (along X) x 2 rows (along Y)
#
# Part.makeOblong behaves exactly like Part.makeBox:
#   makeOblong(p1, p2, p3, pnt, dir)
#   p1, p2, p3 are sizes along 3 local axes determined by dir — NOT fixed to length/width.
# For dir=(0,0,1): local X=p1, local Y=p2, local Z=p3 (depth along cut direction).
#   => p1 = size along X, p2 = size along Y, p3 = cut depth along Z
#   pnt = lower-left corner: pnt.x = center_x - p1/2, pnt.y = center_y - p2/2

base_slot_length    = 10.0             # p1 -> extent along X (long axis)
base_slot_diameter  = 4.0              # p2 -> extent along Y (short axis)
base_slot_cut_depth = thickness + 2.0  # p3 -> cut depth along Z

# --- X positions (2 columns) ---
# 20 mm gap from the far edge (X = dim_x)
# Rightmost center: dim_x - 20 - p1/2 = 100 - 20 - 5 = 75 mm
# Center-to-center along X = p1 + 10mm gap = 10 + 10 = 20 mm
# Left column center: 75 - 20 = 55 mm
base_slot_cx_col = [dim_x - 20.0 - base_slot_length / 2.0,                            # col 1 (right): 75 mm
                    dim_x - 20.0 - base_slot_length / 2.0 - (base_slot_length + 10.0)] # col 0 (left):  55 mm

# --- Y positions (2 rows, centered in width) ---
# Center-to-center along Y = p2 + 5mm gap = 4 + 5 = 9 mm
# Centered in dim_y=30 -> row centers at 15 +/- 4.5 mm
base_slot_cy_row = [dim_y / 2.0 + (base_slot_diameter + 5.0) / 2.0,  # row 1 (top):    19.5 mm
                    dim_y / 2.0 - (base_slot_diameter + 5.0) / 2.0]  # row 0 (bottom):  10.5 mm

# Cut all 4 slots
for cx in base_slot_cx_col:
    for cy in base_slot_cy_row:
        pnt_x = cx - base_slot_length / 2.0    # corner X (pnt = lower-left corner, not center)
        pnt_y = cy - base_slot_diameter / 2.0  # corner Y
        base_slot_tool = Part.makeOblong(
            base_slot_length,    # p1 -> X extent (long)
            base_slot_diameter,  # p2 -> Y extent (short)
            base_slot_cut_depth, # p3 -> Z depth (cut direction)
            App.Vector(pnt_x, pnt_y, 0.0),  # corner at z=0 (bottom face)
            App.Vector(0, 0, 1)             # dir: cut upward through base
        )
        bracket_shape = bracket_shape.cut(base_slot_tool)

# 5. Add oblong slot on VERTICAL WALL (leg2)
#    Centered in width (dim_y), center 10 mm from top edge, oriented along the length
leg2_slot_length = 20.0  # Desired extent along Y (horizontal, dim_y direction)
leg2_slot_width  = 6.0   # Desired extent along Z (vertical, flange height direction)
cut_depth = thickness + 2.0

leg2_slot_center_y = dim_y / 2.0                                    # Centered in width
leg2_slot_center_z = flange_height - 10.0 - leg2_slot_width / 2.0  # Center 10 mm from the top edge

# Part.makeOblong behaves exactly like Part.makeBox:
#   makeOblong(p1, p2, p3, pnt, dir)
# For dir=(1,0,0): p1=Z-extent, p2=Y-extent, p3=X cut depth
# 90° natural-coordinate convention:
#   pnt.y = center_y + size_y/2  (ADD — Y runs in −Y from pnt)
#   pnt.z = center_z − size_z/2  (SUBTRACT — Z runs upward from pnt)
#   pnt.x = 0
leg2_slot_tool = Part.makeOblong(
    leg2_slot_width,   # p1 -> Z extent (short axis, vertical)
    leg2_slot_length,  # p2 -> Y extent (long axis, along dim_y)
    cut_depth,         # p3 -> X depth (cut direction)
    App.Vector(0, leg2_slot_center_y + leg2_slot_length / 2.0, leg2_slot_center_z - leg2_slot_width / 2.0),
    App.Vector(1, 0, 0)
)

# ── Batch-cut all leg2 features in a single call (leg2 face found only once) ──
# Tools use 90° natural-coordinate convention: pnt encodes actual (cy, cz) of the feature.
# map_and_cut_leg2_batch transforms them to match the actual bend angle automatically.
leg2_tools = [
    {"tool_shape": leg2_hole_tool, "hole_center_y": leg2_hole_y, "hole_center_z": leg2_hole_z},
    {"tool_shape": leg2_slot_tool, "hole_center_y": leg2_slot_center_y, "hole_center_z": leg2_slot_center_z},
]

bracket_shape = map_and_cut_leg2_batch(
    bracket_shape=bracket_shape,
    tools=leg2_tools,
    angle_deg=bend_angle_deg,
    dim_y=dim_y,
    thickness=thickness,
    flange_height=flange_height,
    bend_radius=bend_radius
)


# Register final bracket object
final_bracket_obj = doc.addObject("Part::Feature", "FinalBracket")
final_bracket_obj.Shape = bracket_shape
final_bracket_obj.Label = "L-Bracket 120° with Holes and Fillets"
doc.recompute()

# Export files
output_dir = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir, exist_ok=True)

step_path = os.path.join(output_dir, f"{sanitized_title}.step")
Import.export([final_bracket_obj], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(bracket_shape, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    bracket_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)

# Export OBJ with separate meshes
obj_path = os.path.join(output_dir, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [final_bracket_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(Shape=face, LinearDeflection=0.1, AngularDeflection=0.523599)
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)
