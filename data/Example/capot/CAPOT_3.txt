#Example boite avec 4 plis, matiere inox, l'interieur fait 500mm x 500 mm.
# Pli arriere (back) a 90 deg vers le haut et largeur 10mm.
# Pli droit (right) 90 deg vers le haut et largeur 20mm.
# Pli avant (front) 90 deg vers le bas et largeur 10mm.
# Pli gauche (left) 90 deg vers le bas et largeur 20mm.
# Shape type: capot
# Subtype: four-side folded plate / mixed-direction folded box
# Subtype (optional variant): four-side-folded-plate-single-flange-corner-overlap
# Retrieval keywords: boite avec 4 plis, capot mixed direction, plis independants,
# pli arriere et droit vers le haut, pli avant et gauche vers le bas, do not use makeTub,
# 1 pli nhô ra 2 dau/goc, flange overlap symmetric both ends, mep 1 pli chong len 2 goc,
# corner overlap on one flange only.
#
# ⚠️ WALL-NAME STANDARD (applies to EVERY CAPOT script, mixed-direction or not):
# CAPOT walls are ALWAYS named front / back / left / right - the SAME 4 names
# used everywhere else in this codebase (Part.makeTub, AddOutwardBend,
# get_capot_wall_frame, resolve_capot_wall_hole_position in FreeCadUtil all
# ONLY accept these 4 strings and raise ValueError on anything else).
# There is NO "top"/"bottom" wall in a CAPOT. Fixed neighbor pairs (used for
# corner-overlap wording): front <-> {left, right}; back <-> {left, right};
# left <-> {front, back}; right <-> {front, back}.
#
# Notes:
# - Use real SheetMetal bends because each side has its own width and direction.
# - Do not use Part.makeTub for this mixed-direction capot.
# - "L'interieur fait 500mm x 500mm" is the central inside flat reference.
# - Missing thickness is assumed as 1.0 mm for this runnable demo only.
# - Holes/cuts use one operation-card format and a local face mapper.
#
# CORNER OVERLAP FEATURE (read before touching corner_overlap_operations):
# - This uses the SAME "operation-card" list pattern as hole_operations below:
#   a plain list of small dicts, one per flange that needs the treatment.
#   Zero, one, two, three, or all four flanges can each get their own card -
#   this is a general per-flange feature, NOT a single "pick one flange" flag.
# - DEFAULT BEHAVIOR: corner_overlap_operations = [] (empty list = no overlap
#   anywhere). If the user_request does NOT explicitly ask for a flange to
#   overhang / overlap / "nho ra" at a corner, LEAVE THIS EMPTY. Do not
#   invent an overlap that wasn't asked for.
# - Add ONE card per flange the user names, in the same order the user
#   mentioned them:
#     {"flange": "front" | "back" | "left" | "right", "extra": <mm, float>}
#   - 1 flange named ("the left wall overlaps...") -> 1 card.
#   - 2 flanges named, same or different amounts ("left and right overlap by
#     3mm" / "front overlaps by 3mm and back by 5mm") -> 2 separate cards,
#     each with its own "extra" value.
#   - 3+ flanges named, spanning BOTH axes ("overlap left right front") -> one
#     card PER flange named (3 cards here: left, right, front) - NEVER
#     substitute the named flanges for their neighbor pair. Naming "left" and
#     "right" together does NOT mean "front/back overlap the left/right
#     walls" - it means left gets a card AND right gets a card, each
#     overlapping ITS OWN fixed neighbors (front/back).
#   - Every flange named -> up to 4 cards (one per flange). Any flange NOT
#     named by the user gets NO card (its overlap stays 0 - never default an
#     un-mentioned flange to the same "extra" as a mentioned one, and never
#     add a flange - e.g. "back" - that the user did not name).
# - Each flange's overlap is symmetric on both of its own ends
#   (extend1 == extend2 == that card's "extra"). There is no valid variant
#   where only one end of a flange extends and the other stays at 0 - a
#   flange either has a card (both ends extend equally) or it doesn't
#   (both ends stay 0).
# - Implementation uses SMBendWall's OWN native "extend1"/"extend2"
#   properties (App::PropertyDistance, "Extend from Left Side"/"Extend from
#   Right Side"). NEVER hand-roll this by fusing extra Part.makeBox()
#   volumes onto bend_obj.Shape: SMBendWall's returned Shape is the base
#   plate FUSED with the wall (not the wall alone), so any box built from
#   that Shape's BoundBox silently grabs the whole base plate's width and
#   produces a full-width slab instead of a small corner overhang. This was
#   tried and confirmed broken - extend1/extend2 are the only correct way.
# -*- coding: utf-8 -*-
#!/usr/bin/env python3
import sys
import os
import math
import FreeCAD as App
import Part
import Import
import Mesh
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)

from FreeCadUtil import *

# SheetMetal direct import for demo bends.
# No new FreeCadUtil function is created in this example.
sheetmetal_path = os.path.join(project_root, "sheetmetal")
if sheetmetal_path not in sys.path:
    sys.path.insert(0, sheetmetal_path)

import SheetMetalCmd

# 1) Metadata / document setup
sanitized_title = "Boite_4_plis_mixed_direction_demo"  # DO NOT CHANGE THIS VALUE
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# 2) Parameters (mm)
material = "inox"
thickness = 1.0              # Demo assumption: request did not specify thickness
bend_radius = thickness      # Demo assumption: inner bend radius = thickness
inner_x = 500.0              # Interior flat reference along X
inner_y = 500.0              # Interior flat reference along Y
bend_angle = 90.0

back_bend_width = 10.0
right_bend_width = 20.0
front_bend_width = 10.0
left_bend_width = 20.0
bend_length_offset = bend_radius + thickness

# Corner overlap (optional, see "CORNER OVERLAP FEATURE" note above).
# One card per flange that overlaps its neighbors; empty list = no overlap
# anywhere (default). Only add cards for flanges the user_request explicitly
# names.
corner_overlap_operations = [
    # {"flange": "front", "extra": 3.0},
]

# 3) Create the central flat plate.
# Coordinate convention:
#   X from -250 to +250
#   Y from -250 to +250
#   Z from 0 to thickness
base_shape = Part.makeBox(
    inner_x,
    inner_y,
    thickness,
    App.Vector(-inner_x / 2.0, -inner_y / 2.0, 0.0)
)

base_obj = doc.addObject("Part::Feature", "Base_Interior_500x500")
base_obj.Shape = base_shape
base_obj.Label = "Base interior 500x500 inox"
doc.recompute()

# 4) Find the four perimeter edges on the top face of the base plate.
# The helper already exists in FreeCadUtil/CoffreFunction.py.
# CAPOT convention (same as get_capot_wall_frame): front wall = Y=-dim_y/2,
# back wall = Y=+dim_y/2, left wall = X=-dim_x/2, right wall = X=+dim_x/2.
z_edge = thickness
tolerance = 0.1

back_edge, _ = find_edge_by_coordinates(
    base_obj.Shape,
    (-inner_x / 2.0, inner_y / 2.0, z_edge),
    ( inner_x / 2.0, inner_y / 2.0, z_edge),
    tolerance=tolerance
)

right_edge, _ = find_edge_by_coordinates(
    base_obj.Shape,
    (inner_x / 2.0, -inner_y / 2.0, z_edge),
    (inner_x / 2.0,  inner_y / 2.0, z_edge),
    tolerance=tolerance
)

front_edge, _ = find_edge_by_coordinates(
    base_obj.Shape,
    (-inner_x / 2.0, -inner_y / 2.0, z_edge),
    ( inner_x / 2.0, -inner_y / 2.0, z_edge),
    tolerance=tolerance
)

left_edge, _ = find_edge_by_coordinates(
    base_obj.Shape,
    (-inner_x / 2.0, -inner_y / 2.0, z_edge),
    (-inner_x / 2.0,  inner_y / 2.0, z_edge),
    tolerance=tolerance
)

if not all([back_edge, right_edge, front_edge, left_edge]):
    raise RuntimeError("Could not find all four base perimeter edges for bending.")

# 5) Create the four independent bends.
# Existing PlateFunction.py examples subtract bend_radius + thickness from the
# requested flange height. This example keeps the same convention.


def make_sheetmetal_bend(name, edge_name, requested_width, invert, extend1=0.0, extend2=0.0):
    bend_obj = doc.addObject("Part::FeaturePython", name)
    SheetMetalCmd.SMBendWall(bend_obj, base_obj, [edge_name])
    bend_obj.length = requested_width - bend_length_offset
    bend_obj.angle = 180.0 - bend_angle
    bend_obj.radius = bend_radius
    bend_obj.invert = invert
    # Native SheetMetal corner-overlap knobs (both ends always equal - see
    # "CORNER OVERLAP FEATURE" note above). Stay 0.0/0.0 unless this flange
    # has a card in corner_overlap_operations.
    bend_obj.extend1 = extend1
    bend_obj.extend2 = extend2
    doc.recompute()
    return bend_obj


bend_specs = [
    {"name": "SMBend_Back_Up_10mm", "edge": back_edge, "width": back_bend_width, "invert": False, "flange": "back"},
    {"name": "SMBend_Right_Up_20mm", "edge": right_edge, "width": right_bend_width, "invert": False, "flange": "right"},
    {"name": "SMBend_Front_Down_10mm", "edge": front_edge, "width": front_bend_width, "invert": True, "flange": "front"},
    {"name": "SMBend_Left_Down_20mm", "edge": left_edge, "width": left_bend_width, "invert": True, "flange": "left"},
]

# Look up each flange's overlap amount by name; flanges with no card get 0.0.
corner_overlap_by_flange = {op["flange"]: op["extra"] for op in corner_overlap_operations}

bend_objects = [
    make_sheetmetal_bend(
        spec["name"], spec["edge"], spec["width"], spec["invert"],
        extend1=corner_overlap_by_flange.get(spec["flange"], 0.0),
        extend2=corner_overlap_by_flange.get(spec["flange"], 0.0),
    )
    for spec in bend_specs
]

# 6) Fuse the four independently bent results into one demo solid.
# This is acceptable for a demo file. A production utility should bend one
# evolving sheet body and add corner relief handling.
final_shape = bend_objects[0].Shape
for bend_obj in bend_objects[1:]:
    final_shape = final_shape.fuse(bend_obj.Shape)

final_shape = final_shape.removeSplitter()

# 7) Add demo holes on the base and on the four folded faces.
# The important teaching point is the unified hole_operations tag below.
# All faces use the same card structure. The mapper is the only place where
# face-specific world X/Y/Z signs appear.

cut_depth = thickness + 4.0
base_cut_z = -2.0


def base_xy(u, v):
    """Map base local coordinates (u, v) to world X/Y."""
    return (-inner_x / 2.0 + u, -inner_y / 2.0 + v)


def hex_radius_from_op(op):
    if "radius" in op:
        return op["radius"]
    if "flat_diameter" in op:
        return op["flat_diameter"] / math.sqrt(3.0)
    raise ValueError("Hexagon cuts require radius or flat_diameter.")


def make_base_tool(op):
    x, y = base_xy(op["u"], op["v"])
    if op["shape"] == "circle":
        return Part.makeCylinder(op["radius"], cut_depth, App.Vector(x, y, base_cut_z), App.Vector(0, 0, 1))
    if op["shape"] == "hexagon":
        return Part.makeHexagon(hex_radius_from_op(op), cut_depth, App.Vector(x, y, base_cut_z), App.Vector(0, 0, 1))
    if op["shape"] == "square":
        size = op["size"]
        return Part.makeBox(size, size, cut_depth, App.Vector(x - size / 2.0, y - size / 2.0, base_cut_z))
    if op["shape"] == "rectangle":
        size_u = op["size_u"]
        size_v = op["size_v"]
        return Part.makeBox(size_u, size_v, cut_depth, App.Vector(x - size_u / 2.0, y - size_v / 2.0, base_cut_z))
    if op["shape"] == "oblong":
        size_u = op["size_u"]
        size_v = op["size_v"]
        return Part.makeOblong(size_u, size_v, cut_depth, App.Vector(x - size_u / 2.0, y - size_v / 2.0, base_cut_z))
    raise ValueError(f"Unsupported base cut shape: {op['shape']}")


def folded_face_frame(face):
    if face == "back_flange":
        return {
            "axis": App.Vector(0, 1, 0),
            "origin": App.Vector(-inner_x / 2.0, inner_y / 2.0 - cut_depth / 2.0, thickness),
            "u_dir": App.Vector(1, 0, 0),
            "h_dir": App.Vector(0, 0, 1),
            "box_dims": "u_depth_h",
        }
    if face == "right_flange":
        return {
            "axis": App.Vector(1, 0, 0),
            "origin": App.Vector(inner_x / 2.0 - cut_depth / 2.0, -inner_y / 2.0, thickness),
            "u_dir": App.Vector(0, 1, 0),
            "h_dir": App.Vector(0, 0, 1),
            "box_dims": "depth_u_h",
        }
    if face == "front_flange":
        return {
            "axis": App.Vector(0, 1, 0),
            "origin": App.Vector(-inner_x / 2.0, -inner_y / 2.0 - cut_depth / 2.0, 0.0),
            "u_dir": App.Vector(1, 0, 0),
            "h_dir": App.Vector(0, 0, -1),
            "box_dims": "u_depth_h",
        }
    if face == "left_flange":
        return {
            "axis": App.Vector(1, 0, 0),
            "origin": App.Vector(-inner_x / 2.0 - cut_depth / 2.0, -inner_y / 2.0, 0.0),
            "u_dir": App.Vector(0, 1, 0),
            "h_dir": App.Vector(0, 0, -1),
            "box_dims": "depth_u_h",
        }
    raise ValueError(f"Unsupported folded face: {face}")


def make_folded_tool(op):
    frame = folded_face_frame(op["face"])
    center = frame["origin"] + frame["u_dir"] * op["u"] + frame["h_dir"] * op["h"]

    if op["shape"] == "circle":
        return Part.makeCylinder(op["radius"], cut_depth, center, frame["axis"])
    if op["shape"] == "hexagon":
        return Part.makeHexagon(hex_radius_from_op(op), cut_depth, center, frame["axis"])

    if op["shape"] == "square":
        size_u = op["size"]
        size_h = op["size"]
    elif op["shape"] in ("rectangle", "oblong"):
        size_u = op["size_u"]
        size_h = op["size_h"]
    else:
        raise ValueError(f"Unsupported folded cut shape: {op['shape']}")

    if frame["box_dims"] == "u_depth_h":
        pnt = App.Vector(center.x - size_u / 2.0, center.y, center.z - size_h / 2.0)
        if op["shape"] == "oblong":
            return Part.makeOblong(size_u, size_h, cut_depth, pnt, frame["axis"])
        return Part.makeBox(size_u, cut_depth, size_h, pnt)

    pnt = App.Vector(center.x, center.y - size_u / 2.0, center.z - size_h / 2.0)
    if op["shape"] == "oblong":
        return Part.makeOblong(cut_depth, size_u, size_h, pnt, frame["axis"])
    return Part.makeBox(cut_depth, size_u, size_h, pnt)


def make_cut_tool(op):
    if op["face"] == "base":
        return make_base_tool(op)
    return make_folded_tool(op)


hole_operations = [
    # Base demo holes: circle, square, and rectangle.
    {"op": "cut", "face": "base", "shape": "circle", "u": 120.0, "v": 120.0, "radius": 5.0},
    {"op": "cut", "face": "base", "shape": "square", "u": 250.0, "v": 250.0, "size": 10.0},
    {"op": "cut", "face": "base", "shape": "rectangle", "u": 380.0, "v": 380.0, "size_u": 18.0, "size_v": 10.0},

    # Folded-face demo holes.
    # Keep h smaller than the flange width: front/back width = 10 mm, left/right = 20 mm.
    {"op": "cut", "face": "back_flange", "shape": "circle", "u": 160.0, "h": 5.0, "radius": 2.0},
    {"op": "cut", "face": "back_flange", "shape": "rectangle", "u": 340.0, "h": 5.0, "size_u": 12.0, "size_h": 4.0},
    {"op": "cut", "face": "right_flange", "shape": "circle", "u": 180.0, "h": 10.0, "radius": 3.0},
    {"op": "cut", "face": "right_flange", "shape": "square", "u": 330.0, "h": 10.0, "size": 6.0},
    {"op": "cut", "face": "front_flange", "shape": "circle", "u": 160.0, "h": 5.0, "radius": 2.0},
    {"op": "cut", "face": "front_flange", "shape": "rectangle", "u": 340.0, "h": 5.0, "size_u": 12.0, "size_h": 4.0},
    {"op": "cut", "face": "left_flange", "shape": "circle", "u": 180.0, "h": 10.0, "radius": 3.0},
    {"op": "cut", "face": "left_flange", "shape": "square", "u": 330.0, "h": 10.0, "size": 6.0},
]

for op in hole_operations:
    final_shape = final_shape.cut(make_cut_tool(op))

final_shape = final_shape.removeSplitter()

final_obj = doc.addObject("Part::Feature", "Boite_4_Plis_Mixed_Direction")
final_obj.Label = sanitized_title
final_obj.Shape = final_shape
doc.recompute()

# 8) Export directory handling
output_dir_abs = f"./app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

# STEP export
step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([final_obj], step_path)

# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [final_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)

# Final recompute
doc.recompute()
