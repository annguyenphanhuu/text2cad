#Example 1: Right-isosceles triangular sheet-metal part with one hypotenuse flange
# Create a right-isosceles triangular sheet-metal part.
# User phrasing example after clarification: "piece triangulaire rectangle isocele avec un grand cote de 200mm, epaisseur 2mm, pli de 20mm exterieur sur le grand cote".
# IMPORTANT PARAMETER MAPPING:
# - "angle droit" means right triangle.
# - "grand cote", "cote le plus long", "longest side" means the hypotenuse.
# - With an explicitly confirmed right-isosceles triangle, hypotenuse_length = 200 mm and both legs are derived from hypotenuse / sqrt(2).
# GENERATOR DECISION NOTES:
# - If the user says only "angle droit" + "grand cote" / "cote le plus long" / "hypotenuse" and does NOT say "isocele", unified analysis must ask for one perpendicular leg OR confirmation that it is right-isosceles.
# - Use `get_right_isosceles_triangle_points_from_hypotenuse(hypotenuse_length)` only after the user explicitly says or confirms right-isosceles / triangle rectangle isocele.
# - If the user gives two perpendicular legs, use `get_right_triangle_points_from_legs(x_leg_length, y_leg_length)` instead.
# - If the user gives hypotenuse plus one leg, use `get_right_triangle_points_from_hypotenuse_and_leg(hypotenuse_length, leg_length, leg_axis)`.
# - Never pass a "grand cote" value as `x_leg_length` or `y_leg_length`.
# - Bend wording: "sur le grand cote" on a right triangle means the hypotenuse edge B-C, so select points[1] -> points[2].
# Coordinate convention from FreeCadUtil helper:
# - points[0] = A = right-angle vertex at origin.
# - points[1] = B = endpoint of the X leg.
# - points[2] = C = endpoint of the Y leg.
# - The hypotenuse / grand cote is edge B-C, therefore bend selection MUST use points[1] -> points[2].
# Thickness is 2 mm.
# Add one 20 mm outward flange along the hypotenuse only.
# Bend angle is 90 degrees. Bend radius is equal to the thickness.
# Shape type: Triangle
#
# -*- coding: utf-8 -*-
#!/usr/bin/env python3
import os
import sys
import FreeCAD as App
import Part
import Import
import Mesh
import MeshPart

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# 1) Metadata / document setup
sanitized_title = "Triangle_RightIsosceles_Hypotenuse200_Thk2_HypotenuseFlange20"
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# 2) Parameters (mm)
hypotenuse_length = 200.0
thickness = 2.0
flange_height = 20.0
bend_angle_deg = 90.0
bend_radius = thickness


def bootstrap_sheetmetal():
    sheetmetal_path = os.path.join(project_root, "sheetmetal")
    if sheetmetal_path not in sys.path:
        sys.path.insert(0, sheetmetal_path)

    import SheetMetalCmd
    import SheetMetalTools

    def task_restore_defaults(_obj, _default_vars):
        pass

    SheetMetalTools.taskRestoreDefaults = task_restore_defaults
    return SheetMetalCmd


SheetMetalCmd = bootstrap_sheetmetal()

# 3) Create triangular base plate
# GENERATOR DECISION:
# - The confirmed subtype is right-isosceles from hypotenuse.
# - The user's `grand cote` is stored as `hypotenuse_length`.
# - This helper derives both perpendicular legs from the hypotenuse.
triangle_points = get_right_isosceles_triangle_points_from_hypotenuse(hypotenuse_length)
base_shape = make_triangle_plate(triangle_points, thickness)

base_obj = doc.addObject("Part::Feature", "Base_Right_Isosceles_Triangle")
base_obj.Shape = base_shape
base_obj.Label = "Right-isosceles triangle hypotenuse 200 thickness 2"
doc.recompute()

# 4) Add one sheet-metal bend along the grand cote / hypotenuse
# EDGE NAMING FOR RIGHT TRIANGLE:
# - A-B = points[0] -> points[1] = X leg, NOT the grand cote.
# - A-C = points[0] -> points[2] = Y leg, NOT the grand cote.
# - B-C = points[1] -> points[2] = hypotenuse / grand cote.
# Because the user requested the bend on "grand cote", select points[1] -> points[2].
hypotenuse_start = App.Vector(triangle_points[1].x, triangle_points[1].y, thickness)
hypotenuse_end = App.Vector(triangle_points[2].x, triangle_points[2].y, thickness)
hypotenuse_edge_name = find_edge_by_points(base_obj.Shape, hypotenuse_start, hypotenuse_end)
if not hypotenuse_edge_name:
    raise RuntimeError("Could not find the hypotenuse / grand cote edge for the bend.")

bend_obj = doc.addObject("Part::FeaturePython", "SMBend_Hypotenuse_Flange")
SheetMetalCmd.SMBendWall(bend_obj, base_obj, [hypotenuse_edge_name])
# SheetMetal SMBendWall expects the straight flange length after the bend zone.
# Use the same convention as the bracket examples: requested height minus radius and thickness.
bend_obj.length = flange_height - (bend_radius + thickness)
bend_obj.angle = 180.0 - bend_angle_deg
bend_obj.radius = bend_radius
bend_obj.invert = False
bend_obj.AutoMiter = True
bend_obj.minGap = 0.0
bend_obj.maxExtendDist = flange_height
bend_obj.reliefw = 0.0
bend_obj.reliefd = 0.0
doc.recompute()

final_shape = bend_obj.Shape.removeSplitter()

result = doc.addObject("Part::Feature", "TriangleHypotenuseFlange")
result.Label = sanitized_title
result.Shape = final_shape
doc.recompute()

# 5) Export to STEP, geometry JSON, and OBJ
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

step_path = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([result], step_path)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

geometry_result = analyze_geometry(final_shape, material="{mapped_material}")

geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    final_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)

# Export OBJ with separate meshes for each face (MANDATORY PATTERN)
obj_path = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [result]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_path)
doc.recompute()
