# Example Create an L-shaped bracket with a 131 mm base, a height of 57 mm, a width of 80 mm, and a thickness of 6 mm, with a 90° bend and a bending radius of 1 mm on the inside and 7 mm on the outside.
# The base must include four 5 mm diameter holes arranged in a square pattern (2 rows and 2 columns), spaced 40 mm apart, centered in width, and positioned 19 mm from the edge opposite the bend.
# Shape type: L-bracket

import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart
import Draft

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *

# Define the sanitized_title variable  
sanitized_title = "Support_en_L_131x80x6_H57_Rint1_Rext7_4Trous5_Congs10"

# Document Setup
doc = App.newDocument("GeneratedModel")
doc.Label = sanitized_title

# 1. Create the L-shaped bracket
# Parameters (mm)
dim_x         = 131.0   # Base length (X dimension)
dim_y         =  80.0   # Width (Y dimension)
thickness     =   6.0   # Sheet thickness
flange_height =  57.0   # Vertical leg height (Z dimension)
bend_angle    =  90.0   # Bend angle
bend_radius   =   1.0   # Inner bend radius

bracket = Part.makeLShape(
    dim_x=dim_x,
    dim_y=dim_y,
    thickness=thickness,
    flange_height=flange_height,
    bend_angle_deg=bend_angle,
    bend_radius=bend_radius,
)

# 2. Add four D5 holes on horizontal leg (leg1)
# The base must include four 5 mm diameter holes arranged in a square pattern
# (2 rows and 2 columns), spaced 40 mm apart, centered in width, 
# and positioned 19 mm from the edge opposite the bend.

# Hole parameters on horizontal leg (base)
hole_diam     =   5.0
hole_rad      = hole_diam / 2.0
hole_depth    = thickness + 2.0
spacing       =  40.0  # spacing between holes (square pattern)
edge_offset   =  19.0  # positioned 19 mm from the outer edge opposite bend

# 2.1 First column of two holes positioned 19 mm from edge opposite bend
hole_x1 = dim_x - edge_offset
hole1_y = (dim_y - spacing) / 2.0
hole2_y = hole1_y + spacing

# Hole 1
h1 = Part.makeCylinder(
    hole_rad,
    hole_depth,
    App.Vector(hole_x1, hole1_y, -1.0),
    App.Vector(0, 0, 1)
)
bracket = bracket.cut(h1)

# Hole 2
h2 = Part.makeCylinder(
    hole_rad,
    hole_depth,
    App.Vector(hole_x1, hole2_y, -1.0),
    App.Vector(0, 0, 1)
)
bracket = bracket.cut(h2)

# 2.2 Second column of two holes positioned 40mm apart from the first column
hole_x2 = hole_x1 - spacing

# Hole 3
h3 = Part.makeCylinder(
    hole_rad,
    hole_depth,
    App.Vector(hole_x2, hole1_y, -1.0),
    App.Vector(0, 0, 1)
)
bracket = bracket.cut(h3)


# Hole 4
h4 = Part.makeCylinder(
    hole_rad,
    hole_depth,
    App.Vector(hole_x2, hole2_y, -1.0),
    App.Vector(0, 0, 1)
)
bracket = bracket.cut(h4)

final_shape = bracket

# 3. Register final object
final_obj = doc.addObject("Part::Feature", "Support_en_L")
final_obj.Label = sanitized_title
final_obj.Shape = final_shape
doc.recompute()

# Export STEP
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)
step_filename = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([final_obj], step_filename)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(final_shape, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    final_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)


# Export OBJ (Face Mesh Export)
obj_filename = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [final_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599,
        )
        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i+1:02d}"
        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()
Mesh.export(mesh_objects, obj_filename)
