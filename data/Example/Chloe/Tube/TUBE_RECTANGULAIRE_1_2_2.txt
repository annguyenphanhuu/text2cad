#Example I want an open rectangular steel tube, 190 × 70 × 3 mm, with a length of 2980 mm and an opening of 130 mm.
# Shape type: tube

import os
import sys
import math
import FreeCAD as App
import Part
import Mesh
import Import
import MeshPart
import Draft

# MANDATORY PATH SETUP - NEVER SKIP THIS
script_dir = os.path.dirname(os.path.abspath(__file__)) if "__file__" in globals() else os.getcwd()
project_root = os.path.abspath(os.path.join(script_dir, "..", "..", ".."))
if project_root not in sys.path:
    sys.path.insert(0, project_root)
from FreeCadUtil import *


# 1. DOCUMENT INITIALIZATION
doc = App.newDocument("OpenRectangularTube")
sanitized_title = "open_rectangular_steel_tube_190x70x3_L2980_opening_130"
doc.Label = sanitized_title

# 2. CREATE THE RECTANGULAR TUBE
# Coordinate system: Y = length, X = width (190mm), Z = height (70mm)
length       = 2980.0        # Tube length along Y-axis (mm)
outer_width  = 190.0         # Outer width (X dimension) in mm
outer_height = 70.0          # Outer height (Z dimension) in mm
thickness    = 3.0           # Wall thickness in mm

# ── LONG FACE / SHORT FACE IDENTIFICATION ──────────────────────────────────
# long_side  = max(outer_width, outer_height) = max(190, 70) = 190 mm
# short_side = min(outer_width, outer_height) = min(190, 70) =  70 mm
#
# Summary for Part.makeRectangularTube(length, outer_width, outer_height, ...):
#   outer_width  (X) = 190 mm → LONG  side → TOP/BOTTOM face  (Z-max, Z-min) → drill along Z-axis (±Z)
#   outer_height (Z) =  70 mm → SHORT side → FRONT/BACK face  (X-min, X-max) → drill along X-axis (±X)
#
# ⚠️ When description says "Short face (70 mm)"  → FRONT/BACK face → pnt.x=0 or outer_width, dir=(±1,0,0)
# ⚠️ When description says "Long face  (190 mm)" → TOP/BOTTOM face → pnt.z=-2.0 or outer_height, dir=(0,0,±1)
# ──────────────────────────────────────────────────────────────────────────
long_side  = max(outer_width, outer_height)   # = 190 mm — TOP/BOTTOM faces (Z-max, Z-min)
short_side = min(outer_width, outer_height)   # =  70 mm — FRONT/BACK faces (X-min, X-max)

# Automatically apply fillets to the edges — inner_radius = thickness, outer_radius = 2*thickness
tube_shape = Part.makeRectangularTube(length, outer_width, outer_height, thickness, 6.0, 3.0)


# 3. CREATE OPENING CUT ON TOP FACE (Z-max)
# Opening width = 130 mm, full length along Y, cuts through top wall thickness
opening_width = 130.0
cut_depth     = thickness * 1.2   # ensure full penetration through top wall

# 3.1 Position the cut box centered on X (outer_width direction)
#     The box spans the full tube length in Y, opening_width in X, cut_depth in Z
cut_x = (outer_width - opening_width) / 2.0   # X offset: center the opening
cut_y = 0.0                                    # from Y=0 to Y=length (full length)
cut_z = outer_height - thickness               # top surface of top wall

cut_box = Part.makeBox(
    opening_width,   # X extent
    length,          # Y extent (full length)
    cut_depth,       # Z extent (through top wall)
    App.Vector(cut_x, cut_y, cut_z)
)

# Perform the cut to create the opening
tube_with_opening = tube_shape.cut(cut_box)
final_shape = tube_with_opening

# 4. FINALIZE OBJECT IN DOCUMENT
tube_obj = doc.addObject("Part::Feature", "RectangularTube")
tube_obj.Shape = final_shape
tube_obj.Label = sanitized_title
doc.recompute()

# 5. EXPORT STEP
output_dir_abs = f"/app/storage/{sanitized_title}/output"
os.makedirs(output_dir_abs, exist_ok=True)

step_filename = os.path.join(output_dir_abs, f"{sanitized_title}.step")
Import.export([tube_obj], step_filename)

# ============================================================
# GEOMETRY ANALYSIS
# ============================================================
from FreeCadUtil.GeometryAnalyzer import analyze_geometry, export_geometry_to_json

# Analyze final shape geometry
geometry_result = analyze_geometry(final_shape, material="{mapped_material}")


# Export geometry data to JSON
geometry_json = os.path.join(output_dir_abs, f"{sanitized_title}_geometry.json")
export_geometry_to_json(
    final_shape,
    geometry_json,
    material="{mapped_material}",
    additional_info={
        "part_name": sanitized_title,
    }
)


# 6. EXPORT OBJ (Face Mesh Export)
obj_filename = os.path.join(output_dir_abs, f"{sanitized_title}.obj")

mesh_objects = []
face_counter = 1

for part in [tube_obj]:
    shape = part.Shape
    part_name = part.Label

    for i, face in enumerate(shape.Faces):
        face_mesh = MeshPart.meshFromShape(
            Shape=face,
            LinearDeflection=0.1,
            AngularDeflection=0.523599,
        )

        mesh_obj = doc.addObject("Mesh::Feature", f"Face_{face_counter:02d}")
        mesh_obj.Mesh = face_mesh
        mesh_obj.Label = f"{part_name}_Face_{i + 1:02d}"

        mesh_objects.append(mesh_obj)
        face_counter += 1

doc.recompute()

Mesh.export(mesh_objects, obj_filename)
